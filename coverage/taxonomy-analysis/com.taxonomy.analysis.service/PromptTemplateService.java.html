package com.taxonomy.analysis.service;

/**
 * Manages prompt templates for LLM taxonomy analysis.
 * Loads default templates from {@code classpath:prompts/*.txt} at startup
 * and supports runtime overrides without redeployment.
 */
@Service
public class PromptTemplateService {

    public static final String NODE_ASSESSMENT_POLICY = "requirement-node-assessment-v3";
    public static final String FUNCTIONAL_SCOPE_POLICY = "functional-necessity-v2";
    public static final String ROOT_RELEVANCE_POLICY = "root-relevance-v3";

    private static final Pattern PLACEHOLDER = Pattern.compile("\\{\\{[A-Z_]+\\}\\}");

    private static final Logger log = LoggerFactory.getLogger(PromptTemplateService.class);
    private static final Set<String> HISTORICAL_CODES = Set.of("reformulation-node", "reformulation-reconcile-v2");
    private static final Set<String> RELATION_CODES = Set.of("relation-sources", "relation-navigate", "relation-verify");
    private static final Set<String> INDEPENDENT_CODES = Set.of(
            "requirement-intake", "architecture-dialogue", "architecture-focus", "mail-analysis-command");

    /**
     * Categorizes prompt templates into logical families.
     */
    public enum PromptCategory {
        /** Scoring prompts: distribute parent relevance score across children. */
        SCORING,
        /** Extraction prompts: extract requirement candidates from document text. */
        EXTRACTION,
        /** Regulation-mapping prompts: map regulations to architecture taxonomy nodes. */
        REGULATION_MAPPING,
        /** Justification prompts: justify leaf-node classification. */
        JUSTIFICATION
    }

    /** File-based defaults loaded at startup (key = taxonomy code or "default"). */
    private final Map<String, String> defaults = new HashMap<>();

    private final PromptTemplateOverrideStore overrideStore;

    /** Standalone analysis, without application storage. */
    public PromptTemplateService() {
        this(Optional.empty());
    }

    @Autowired
    public PromptTemplateService(Optional<PromptTemplateOverrideStore> overrideStore) {
        this.overrideStore = overrideStore.orElseGet(StandaloneOverrides::new);
    }

    @PostConstruct
    public void loadDefaults() {
        PathMatchingResourcePatternResolver resolver = new PathMatchingResourcePatternResolver();
        try {
            Resource[] resources = resolver.getResources("classpath:prompts/*.txt");
            for (Resource resource : resources) {
                String filename = resource.getFilename();
                if (filename == null) continue;
                String code = filename.endsWith(".txt")
                        ? filename.substring(0, filename.length() - 4)
                        : filename;
                try (InputStream is = resource.getInputStream()) {
                    String content = new String(is.readAllBytes(), StandardCharsets.UTF_8);
                    defaults.put(code, content);
                    log.debug("Loaded prompt template: {}", filename);
                } catch (IOException e) {
                    log.warn("Failed to load prompt template '{}': {}", filename, e.getMessage());
                }
            }
            log.info("Loaded {} prompt template(s) from classpath:prompts/", defaults.size());
        } catch (IOException e) {
            log.warn("Could not scan classpath:prompts/ for templates: {}", e.getMessage());
        }
    }

    /** Reads the shipped text without changing whitespace or consulting runtime overrides. */
    public static String shippedTemplate(String code) {
        try (InputStream stream = PromptTemplateService.class.getResourceAsStream("/prompts/" + code + ".txt")) {
            if (stream == null) throw new IllegalStateException("Missing shipped prompt template: " + code);
            return new String(stream.readAllBytes(), StandardCharsets.UTF_8);
        } catch (IOException failure) {
            throw new IllegalStateException("Could not read shipped prompt template: " + code, failure);
        }
    }

    /**
     * Returns the effective template for a taxonomy code:
     * override → taxonomy-specific default → "default" fallback.
     */
    public String getTemplate(String taxonomyCode) {
        String override = overrideStore.snapshot().get(taxonomyCode);
        return override != null ? override : getDefaultTemplate(taxonomyCode);
    }

    /**
     * Always returns the file-based default (ignoring overrides).
     * Falls back to "default" template if no taxonomy-specific file exists.
     */
    public String getDefaultTemplate(String taxonomyCode) {
        if (defaults.containsKey(taxonomyCode)) {
            return defaults.get(taxonomyCode);
        }
        return defaults.getOrDefault("default", "");
    }

    /** Stores a runtime override for the given taxonomy code. */
    public void setTemplate(String taxonomyCode, String template) {
        requireEditable(taxonomyCode);
        if (template == null || template.isBlank()) throw new IllegalArgumentException("Prompt template is blank");
        overrideStore.save(taxonomyCode, template);
    }

    /** Removes the runtime override; subsequent calls revert to the file-based default. */
    public void resetTemplate(String taxonomyCode) {
        overrideStore.reset(taxonomyCode);
    }

    /** Returns {@code true} if a runtime override exists for this taxonomy code. */
    public boolean isOverridden(String taxonomyCode) {
        return overrideStore.snapshot().containsKey(taxonomyCode);
    }

    public boolean isEditable(String code) {
        return !HISTORICAL_CODES.contains(code);
    }

    public Map<String, String> overrideSnapshot() {
        return overrideStore.snapshot();
    }

    /** Captures all effective texts from one revision of the override store. */
    public Map<String, String> effectiveTemplates() {
        return effectiveTemplates(overrideSnapshot());
    }

    public Map<String, String> effectiveTemplates(Map<String, String> overrides) {
        var templates = new TreeMap<>(defaults);
        templates.putAll(overrides);
        return Collections.unmodifiableMap(templates);
    }

    /**
     * Keeps scoring checkpoint identity independent of relation guidance.
     * Portfolio analyses also account for the shipped relation policy and customized relation guidance.
     */
    public Map<String, String> analysisFingerprintTemplates(boolean includeRelationOverrides) {
        var templates = new TreeMap<>(effectiveTemplates());
        INDEPENDENT_CODES.forEach(templates::remove);
        templates.put(NODE_ASSESSMENT_POLICY, nodeAssessmentContract());
        if (includeRelationOverrides) {
            templates.put("relation-prompt-policy", RelationSearchProtocol.promptPolicyFingerprint());
        }
        for (String code : RELATION_CODES) {
            if (!includeRelationOverrides || Objects.equals(templates.get(code), defaults.get(code))) {
                templates.remove(code);
            }
        }
        return Collections.unmodifiableMap(templates);
    }

    private void requireEditable(String code) {
        if (!isEditable(code)) throw new IllegalArgumentException("Historical prompt templates cannot be edited");
    }

    /**
     * Renders the effective prompt by substituting all {@code {{...}}} placeholders.
     *
     * @param taxonomyCode the taxonomy root code (e.g. "BP")
     * @param businessText the user-entered business requirement text
     * @param nodeList     the formatted list of taxonomy nodes
     * @param parentScore  the parent node's score (100 for root-level nodes); used for distribution
     * @param expectedKeys comma-separated list of the exact node codes the LLM must use as JSON keys
     * @return the rendered prompt string ready to send to the LLM
     */
    public String renderPrompt(String taxonomyCode, String businessText, String nodeList,
                               double parentScore, String expectedKeys) {
        String name = TAXONOMY_NAMES.getOrDefault(taxonomyCode, taxonomyCode);
        String template = getTemplate(taxonomyCode);
        return substitute(template, Map.of(
                "{{BUSINESS_TEXT}}", businessText,
                "{{NODE_LIST}}", nodeList,
                "{{TAXONOMY_NAME}}", name,
                "{{PARENT_SCORE}}", com.taxonomy.dto.AnalysisPercentages.canonical(parentScore),
                "{{EXPECTED_KEYS}}", expectedKeys != null ? expectedKeys : "")) + numericContract()
                + childAllocationContract(parentScore) + functionalScopeContract() + reasonContract()
                + nodeAssessmentContract() + AnalysisLanguageContext.promptContract();
    }

    /** Evaluates one catalogue root independently, without a child allocation budget. */
    public String renderRootPrompt(String taxonomyCode, String businessText, String nodeList) {
        String template = getTemplate("root-relevance");
        return substitute(template, Map.of(
                "{{BUSINESS_TEXT}}", businessText,
                "{{NODE_LIST}}", nodeList,
                "{{TAXONOMY_NAME}}", TAXONOMY_NAMES.getOrDefault(taxonomyCode, taxonomyCode),
                "{{EXPECTED_KEYS}}", taxonomyCode)) + numericContract() + "This root is assessed independently, not normalized across taxonomy roots.\n"
                + functionalScopeContract() + rootRelevanceContract() + reasonContract()
                + nodeAssessmentContract() + AnalysisLanguageContext.promptContract();
    }

    /** Root relevance has an explicit undecided outcome, unlike an allocated child budget. */
    public static String rootRelevanceContract() {
        return "\nMandatory root relevance policy (" + ROOT_RELEVANCE_POLICY + "): "
                + "Assess whether this catalogue domain contributes to the requested function, including necessary enabling services. "
                + "A contribution may be supplied by an existing, shared or externally operated service; relevance does not require building or owning a new service. "
                + "Do not reject a needed dependency merely because the requirement does not ask to implement that dependency again. "
                + "RELEVANT requires a supported functional contribution, with a finite score greater than 0 and at most 100. "
                + "IRRELEVANT requires an evidenced mismatch or exclusion, with score 0. "
                + "If missing facts or unresolved implementation alternatives prevent that decision, use UNRESOLVED with score null and one concrete clarification question. "
                + "Not unconditionally necessary does not mean irrelevant. Do not turn missing detail into a zero, an invented positive score or a confidence percentage. "
                + "A supported broad contribution may be relevant while its particular technology remains unspecified; do not invent that technology. "
                + "For every typed root reply use exactly status, score, reason, question and assessments; give a nonempty reason (at most 2000 characters). "
                + "For RELEVANT and IRRELEVANT set question to an empty string; for UNRESOLVED give a nonempty question (at most 1000 characters). "
                + "The null score exception for UNRESOLVED overrides the numeric-only wording above. "
                + "Do not infer a deployed relationship from a root score.\n";
    }

    /** Shared scope for relevance, source contributions and first-hop relation decisions. */
    public static String functionalScopeContract() {
        return "\nFunctional scope policy (" + FUNCTIONAL_SCOPE_POLICY + "): "
                + "Judge meaning across languages. Scoped contributions may be requested or necessary without literal service names. "
                + "Explain which requested function needs each using the catalogue. If an allowed implementation works without a contribution, "
                + "it is not unconditionally required. Catalogue availability, customary design or generic usefulness is insufficient. "
                + "Scope necessity to function, not vendor. Preserve exclusions, conditions, alternatives and open choices; "
                + "missing information is not rejection. Keep schema and numeric rules. "
                + "A requirement-to-node contribution or a score does not prove a directed architecture edge. "
                + "Verify each edge's direction, function and conditions against its own evidence. "
                + "Existing context and generated relation hypotheses are not new requirements.\n";
    }

    static String nodeAssessmentContract() {
        return """

                Mandatory passage evidence policy (%s).
                Keep each existing offered node ID and its score and reason. Add "assessments" to each node object,
                alongside its existing fields. This overrides any older response example that omits that field.
                Use [] if there is no meaningful passage to assess. Use up to 8 distinct claims and keep claims
                for separate passages or different contributions separate. Give each claim exactly these keys:
                - requirementQuote: a verbatim substring of the original Business Requirement (up to 8000 characters).
                  Preserve spelling, punctuation, spacing and language. Never quote catalogue descriptions, generated
                  summaries or translations as requirement evidence. Use "" only for unresolved or neutral claims that assert no fact.
                  EXPLICIT evidence always requires a quote.
                - subjectKinds: one or more distinct values from PRODUCT, SERVICE, PROCESS, INFORMATION, ROLE, CAPABILITY, UNSPECIFIED.
                  Classify what the original passage is about. Its subject may differ from the offered catalogue node's kind.
                - requirementKinds: one or more distinct values from FUNCTIONAL, QUALITY, CONSTRAINT, ORGANIZATIONAL, CONTEXT, GOAL, UNSPECIFIED.
                - contributionType: choose the direction expressed in the original passage.
                  DIRECT_SUBJECT: the passage describes the node's own function or activity.
                  PROVIDES: the requested product or service supplies the node's function.
                  SUPPORTS: the request enables a downstream activity without specifying its execution.
                  REQUIRES: the requested outcome depends on the node's function.
                  CONSTRAINS: the passage limits the node's behaviour or permitted implementation.
                  UNDETERMINED: the passage does not establish how the request and the node relate.
                - evidenceStatus: SUPPORTED, UNRESOLVED, CONFLICT, NOT_APPLICABLE or OUT_OF_SCOPE.
                - evidenceBasis: EXPLICIT, INFERRED or UNSPECIFIED. This describes the relationship to this exact node,
                  not merely whether the quotation is explicit. Use INFERRED when moving from a stated outcome to a
                  possible implementation, a specialized workflow, an actor-facing application or an incident context.
                - necessity: REQUIRED, OPTIONAL, ALTERNATIVE, CONTEXT or UNDETERMINED.
                  REQUIRED can be conditional. An obligation triggered "bei Bedarf in Abstimmung" remains REQUIRED once
                  the condition applies. Record the condition. Use OPTIONAL when delivery is at the implementer's discretion.
                - scope: DELIVERY, EXISTING_ENVIRONMENT, UNSPECIFIED or EXCLUDED.
                - condition: the applicability condition, or "" when none (up to 1000 characters).
                - rationale: a concise nonempty explanation connecting the passage and the node (up to 2000 characters).
                - question: one concrete question for UNRESOLVED, "" for every other status (up to 1000 characters).
                SUPPORTED requires a quoted passage, a determined contribution and evidence basis. An inferred, indirect
                contribution can be supported. A direct match can be unresolved. Supported OPTIONAL or ALTERNATIVE claims
                require their condition. CONFLICT requires a quoted passage and an explanation of the conflict.
                A shared or existing service may support delivery while remaining EXISTING_ENVIRONMENT. Avoid inferring
                requested delivery from existing infrastructure, conditions, exclusions or implementation choices.
                Bind necessity and delivery scope to the specific node contribution, not to the quoted obligation alone.
                A required outcome does not make every plausible supporting application or process a required deliverable.
                Identify the matched subfunction in the rationale. Do not import other functions, actors, triggers or
                lifecycle obligations from the catalogue definition. For example, a traceable process does not by itself
                establish a mandatory user-operated application; generic recovery does not establish an incident workflow.
                If the source establishes only a possible implementation or downstream use, retain that limitation with
                INFERRED evidence, UNDETERMINED/CONTEXT necessity and UNSPECIFIED/EXISTING_ENVIRONMENT scope as appropriate.
                Use UNRESOLVED with a concrete question when that missing distinction affects the proposed requirement.
                Interpret necessity and evidence independently of the ranking score. Several contributions can all be REQUIRED.
                A low score is insufficient evidence of conflict. A node contribution does not verify an architecture edge.
                """.formatted(NODE_ASSESSMENT_POLICY);
    }

    static String childAllocationContract(double parentScore) {
        return "\nMandatory child allocation policy (child-budget-v3): this overrides conflicting allocation instructions above. "
                + "When at least one offered child has a supported match, the scores must sum to "
                + com.taxonomy.dto.AnalysisPercentages.canonical(parentScore) + ". "
                + "When the parent fits but no offered child has a supported match, return explicit zero for every offered child "
                + "with a specific mismatch reason; the parent assessment remains valid at its supported level. "
                + "Never invent a more specific role, function or implementation merely to spend the inherited budget. "
                + "A child score is an allocated share of that budget, not a confidence or implementation-coverage percentage. "
                + "Preserve a positive child below 1%; for a 1% budget, two equal children have 0.5 each.\n";
    }

    /**

     * @param businessText the user-entered business requirement text
     * @param nodeList     the formatted list of taxonomy nodes
     * @param parentScore  the parent node's score (100 for root-level nodes); used for distribution
     * @return the rendered prompt string ready to send to the LLM

    public String renderPrompt(String taxonomyCode, String businessText, String nodeList, double parentScore) {
        return renderPrompt(taxonomyCode, businessText, nodeList, parentScore, "");
    }

    /**
     * Renders the effective prompt using a parent score of 100 (root-level default).
     * Delegates to {@link #renderPrompt(String, String, String, double, String)}.
     *
     * @param taxonomyCode the taxonomy root code (e.g. "BP")
     * @param businessText the user-entered business requirement text
     * @param nodeList     the formatted list of taxonomy nodes
     * @return the rendered prompt string ready to send to the LLM

    public String renderPrompt(String taxonomyCode, String businessText, String nodeList) {
        return renderPrompt(taxonomyCode, businessText, nodeList, 100, "");
    }

    /**
     * Renders the independent concrete-product suitability prompt. Product scores are not
     * normalized to a parent budget and may all be zero.
     */
    public String renderProductPrompt(String businessText, String nodeList,
                                      String expectedKeys, int minimumScore) {
        String template = getTemplate("IP-product");
        return substitute(template, Map.of(
                "{{BUSINESS_TEXT}}", businessText,
                "{{NODE_LIST}}", nodeList,
                "{{EXPECTED_KEYS}}", expectedKeys != null ? expectedKeys : "",
                "{{MIN_SCORE}}", String.valueOf(minimumScore))) + numericContract() + "Product suitability is independent of parent relevance; do not divide the category budget between products.\n"
                + functionalScopeContract() + reasonContract()
                + nodeAssessmentContract() + AnalysisLanguageContext.promptContract();
    }

    private static String reasonContract() {
        var length = AnalysisReasonContext.effective(AnalysisReasonContext.current());
        String scope = switch (length) {
            case SHORT -> "Write one concise sentence per decision, at most 35 words, naming the decisive match or mismatch.";
            case STANDARD -> "Write 1–2 sentences per decision, at most 60 words, connecting the requirement to the decisive catalogue criterion and assigned relevance.";
            case DETAILED -> "Write 3–5 substantive sentences per decision, at most 100 words. Identify the relevant requirement facts, explain how the catalogue criterion fits or fails to fit them, and justify the assigned weight or suitability. Mention a material limitation or uncertainty only when the supplied evidence supports it.";
        };
        return "\nMandatory decision reason policy (" + AnalysisReasonContext.POLICY + "): " + length.value() + ".\n"
                + "This policy governs every decision reason and overrides different length instructions or shorter illustrative examples above. "
                + scope + " Base the explanation only on supplied requirement and catalogue evidence; do not invent facts, fill gaps with assumptions, or add repetitive filler. "
                + "An explicit zero needs a concrete reason for the mismatch; missing evidence is not proof of irrelevance. "
                + "Keep all requested keys, numeric scoring rules and the required response format.\n";
    }

    private static String numericContract() {
        return "\nMandatory numeric output contract (" + com.taxonomy.dto.AnalysisPercentages.POLICY + "): "
                + "this overrides older integer-only score instructions. Return JSON numbers with decimals allowed, "
                + "between 0 and 100. Use a decimal point, not a comma or a percent sign in the number. "
                + "0.5 means 0.5%, not 50%. Do not round a positive contribution to zero. "
                + "Explicit zero is negative evidence; missing or failed assessment is not zero.\n";
    }

    /**
     * Renders the leaf-justification prompt by substituting all placeholders.
     *
     * @param businessText   the original business requirement text
     * @param leafCode       the code of the leaf node to justify
     * @param pathDescription formatted path from root to leaf with scores and inline reasons
     * @param crossRefs       formatted list of other high-scoring nodes for cross-references
     * @return the rendered prompt string ready to send to the LLM
     */
    public String renderLeafJustificationPrompt(String businessText, String leafCode,
                                                String pathDescription, String crossRefs) {
        String template = overrideSnapshot().getOrDefault("justify-leaf", defaults.getOrDefault("justify-leaf", ""));
        if (template.isBlank()) {
            // Fallback inline template if the file is missing
            template = "You are an expert in C3 taxonomy classification.\n"
                    + "Explain why the taxonomy path ending at {{LEAF_CODE}} "
                    + "best matches the following business requirement.\n\n"
                    + "Business Requirement: {{BUSINESS_TEXT}}\n\n"
                    + "Selected path (root → leaf) with scores:\n{{PATH_DESCRIPTION}}\n"
                    + "Other high-scoring nodes for cross-reference:\n{{CROSS_REFERENCES}}\n\n"
                    + "Provide a coherent justification that explains why this path was chosen, "
                    + "how the leaf node relates to the requirement, and note any relevant "
                    + "connections to the cross-referenced nodes.";
        }
        return substitute(template, Map.of(
                "{{BUSINESS_TEXT}}", businessText,
                "{{LEAF_CODE}}", leafCode,
                "{{PATH_DESCRIPTION}}", pathDescription,
                "{{CROSS_REFERENCES}}", crossRefs)) + reasonContract() + AnalysisLanguageContext.promptContract();
    }

    /**
     * Returns a list of all known template codes (both defaults and any that only have overrides).
     */
    public List<String> getAllTemplateCodes() {
        return List.copyOf(effectiveTemplates().keySet());
    }

    private static final class StandaloneOverrides implements PromptTemplateOverrideStore {
        private final Map<String, String> values = new ConcurrentHashMap<>();

        @Override public Map<String, String> snapshot() { return Map.copyOf(values); }

        @Override public void save(String code, String template) { values.put(code, template); }

        @Override public void reset(String code) { values.remove(code); }
    }

    /** Returns the human-readable taxonomy name for a code, or the code itself if unknown. */
    public String getTaxonomyName(String taxonomyCode) {
        return TAXONOMY_NAMES.getOrDefault(taxonomyCode, taxonomyCode);
    }

    // ── Prompt categories ─────────────────────────────────────────────────────

    /**
     * Returns all template codes for a specific category.
     */
    public List<String> getTemplateCodesByCategory(PromptCategory category) {
        return getAllTemplateCodes().stream()
                .filter(code -> categorizeCode(code) == category)
                .toList();
    }

    /**
     * Determines the {@link PromptCategory} for a given template code.
     */
    public PromptCategory categorizeCode(String code) {
        if (code.startsWith("extract-")) return PromptCategory.EXTRACTION;
        if (code.startsWith("reg-map-")) return PromptCategory.REGULATION_MAPPING;
        if ("justify-leaf".equals(code)) return PromptCategory.JUSTIFICATION;
        return PromptCategory.SCORING;
    }

    // ── Extraction prompt rendering ───────────────────────────────────────────

    /**
     * Renders an extraction prompt for document text.
     *
     * @param extractionCode the extraction template code (e.g. "extract-default")
     * @param documentText   the raw document text to analyze
     * @return the rendered prompt string ready to send to the LLM
     */
    public String renderExtractionPrompt(String extractionCode, String documentText) {
        String template = getTemplate(extractionCode);
        return template.replace("{{DOCUMENT_TEXT}}", documentText);
    }

    // ── Regulation-mapping prompt rendering ───────────────────────────────────

    /**
     * Renders a regulation-to-architecture mapping prompt.
     *
     * @param mappingCode  the mapping template code (e.g. "reg-map-default")
     * @param documentText the raw regulation text
     * @param nodeList     the formatted list of taxonomy nodes
     * @return the rendered prompt string ready to send to the LLM
     */
    public String renderRegulationMappingPrompt(String mappingCode,
                                                 String documentText,
                                                 String nodeList) {
        String template = getTemplate(mappingCode);
        return substitute(template, Map.of("{{DOCUMENT_TEXT}}", documentText, "{{NODE_LIST}}", nodeList));
    }

    /** Match only the original template. Inserted data is literal, including dollar signs,
     * backslashes and text that looks like another placeholder; unknown tokens stay unchanged. */
    private static String substitute(String template, Map<String, String> values) {
        return PLACEHOLDER.matcher(template).replaceAll(match ->
                Matcher.quoteReplacement(values.getOrDefault(match.group(), match.group())));
    }

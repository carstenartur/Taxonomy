package com.taxonomy.export;

import java.awt.geom.Rectangle2D;

/** Produces a standalone SVG with compact shapes and a visible, complete evidence inventory. */
public class SvgDiagramRenderer {
    private final ExportThemeProvider themes;

    public SvgDiagramRenderer() { this(ExportThemeProvider.defaults()); }

    public SvgDiagramRenderer(ExportThemeProvider themes) {
        this.themes = Objects.requireNonNull(themes, "themes");
    }
    private static final int EVIDENCE_FONT_SIZE = 14;
    private static final double EVIDENCE_COLUMN_WIDTH = 720;
    private static final double EVIDENCE_TEXT_WIDTH = EVIDENCE_COLUMN_WIDTH - 28;
    private static final double EVIDENCE_LINE_HEIGHT = 20;

    public String render(DiagramScene scene, DiagramExportContext context, Map<String, String> nodeReasons,
                         Map<String, Map<String, String>> relationReasons, Locale locale) {

    }

    private String render(DiagramScene scene, Map<String, String> nodeReasons,
                          Map<String, Map<String, String>> relationReasons, Locale locale,
                          DiagramCanonicalEvidence canonical, DiagramExportContext context) {
        var language = DiagramLanguage.from(locale);
        if (scene == null || scene.isEmpty())
            throw new IllegalArgumentException("Architecture scene must contain at least one node");
        ExportTheme theme = Objects.requireNonNull(themes.current(), "export theme");
        var markers = new LinkedHashMap<String, String>();
        var detours = new LinkedHashMap<DiagramSceneEdge, DiagramEdgeRouting.Detour>();
        var routedColors = new HashSet<String>();
        var routing = new DiagramEdgeRouting(scene);
        for (var edge : scene.edges()) {
            String color = theme.relationshipStroke(edge.relationType(), proposed(edge, canonical));
            markers.computeIfAbsent(color, ignored -> "arrow-" + (markers.size() + 1));
            var detour = routing.detourForLine(edge);
            if (detour != null) {
                detours.put(edge, detour);
                routedColors.add(color);
            }
        }
        var nodeReferences = new LinkedHashMap<String, String>();
        scene.nodes().forEach(node -> nodeReferences.putIfAbsent(node.id(), "evidence-node-" + (nodeReferences.size() + 1)));
        var edgeReferences = new LinkedHashMap<String, String>();
        scene.edges().forEach(edge -> edgeReferences.putIfAbsent(edge.id(), "evidence-edge-" + (edgeReferences.size() + 1)));
        var evidence = evidence(scene, nodeReasons == null ? Map.of() : nodeReasons,
                relationReasons == null ? Map.of() : relationReasons, nodeReferences, edgeReferences, language, canonical);
        // Grow the sheet in both dimensions instead of making a narrow strip or shrinking the type.
        int columns = Math.max(1, (int) Math.ceil(Math.sqrt(
                evidence.size() * EVIDENCE_LINE_HEIGHT * Math.sqrt(2) / EVIDENCE_COLUMN_WIDTH)));
        int linesPerColumn = (evidence.size() + columns - 1) / columns;
        double width = Math.max(scene.width(), columns * EVIDENCE_COLUMN_WIDTH);
        var summary = new ArrayList<String>();
        for (String line : DiagramPresentation.inventorySummary(scene, language, canonical)) summary.addAll(
                DiagramTextWrapper.wrapMeasured(line, font(11, false), width - 28, Integer.MAX_VALUE, ""));
        String title = DiagramExportContext.title(scene.title(), locale);
        var titleLines = DiagramTextWrapper.wrapMeasured(title, font(18, true), width - 28, Integer.MAX_VALUE, "");
        var headerLinesText = new ArrayList<String>();
        for (String line : context.header()) headerLinesText.addAll(
                DiagramTextWrapper.wrapMeasured(line, font(12, false), width - 28, Integer.MAX_VALUE, ""));
        double graphOffset = 24 + titleLines.size() * 24 + headerLinesText.size() * 18;
        var warningLines = new ArrayList<String>();
        for (String warning : context.warnings()) warningLines.addAll(
                DiagramTextWrapper.wrapMeasured(warning, font(EVIDENCE_FONT_SIZE, false), width - 28, Integer.MAX_VALUE, ""));
        double footer = graphOffset + Math.max(scene.height(), scene.nodes().stream().mapToDouble(n -> n.y() + n.height() + 24).max().orElse(0));
        double warningTop = footer + summary.size() * 16 + 32;
        double evidenceTitleY = footer + summary.size() * 16 + 40
                + (warningLines.isEmpty() ? 0 : 36 + warningLines.size() * EVIDENCE_LINE_HEIGHT);
        var columnHeaders = new ArrayList<List<String>>();
        int headerLines = 1;
        for (int column = 0; column < columns; column++) {
            int firstLine = column * linesPerColumn;
            int end = Math.min(evidence.size(), firstLine + linesPerColumn);
            while (firstLine + 1 < end && evidence.get(firstLine).value().isEmpty()) firstLine++;
            var first = evidence.get(firstLine);
            var header = DiagramTextWrapper.wrapMeasured(language.text("Evidence ", "Evidenz ") + (column + 1) + "/" + columns
                            + " | " + (first.kind().equals("node") ? language.text("node", "Knoten") : language.text("relationship", "Beziehung"))
                            + " " + first.id() + (first.first() ? "" : language.text(" (continued)", " (Fortsetzung)")),
                    font(EVIDENCE_FONT_SIZE, true), EVIDENCE_TEXT_WIDTH, Integer.MAX_VALUE, language.text("Evidence", "Evidenz"));
            columnHeaders.add(header);
            headerLines = Math.max(headerLines, header.size());
        }
        double evidenceTop = evidenceTitleY + 36 + headerLines * EVIDENCE_LINE_HEIGHT;
        double height = evidenceTop + linesPerColumn * EVIDENCE_LINE_HEIGHT + 20;
        StringBuilder svg = new StringBuilder(16_384);
        svg.append("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n")
            .append("<svg xmlns=\"http://www.w3.org/2000/svg\" xmlns:xlink=\"http://www.w3.org/1999/xlink\" role=\"img\" xml:lang=\"").append(language.tag()).append("\" viewBox=\"0 0 ")
            .append(number(width)).append(' ').append(number(height)).append("\" width=\"")
            .append(number(width)).append("\" height=\"").append(number(height)).append("\">\n")
            .append("<title>").append(xml(title)).append("</title>\n")
            .append("<desc>").append(canonical != null
                    ? language.text("Canonical architecture diagram. Shapes link to the visible inventory of complete labels and saved model states.", "Kanonisches Architekturdiagramm. Die Elemente verweisen auf das Inventar mit vollständigen Beschriftungen und gespeicherten Modellzuständen.")
                    : language.text(
                    "Requirement-derived architecture diagram. Shapes link to the visible inventory of complete labels, scores and recorded explanations below. AI verification is not human approval.",
                    "Anforderungsbezogenes Architekturdiagramm. Die Elemente verweisen auf das Inventar mit vollständigen Bezeichnungen, Bewertungen und gespeicherten Begründungen. Eine KI-Prüfung ist keine menschliche Freigabe.")).append("</desc>\n")
            .append("<defs>");
        markers.forEach((color, id) -> {
            svg.append("<marker id=\"").append(id)
                    .append("\" markerWidth=\"10\" markerHeight=\"8\" refX=\"9\" refY=\"4\" orient=\"auto\"><path d=\"M0,0 L10,4 L0,8 z\" fill=\"")
                    .append(color).append("\"/></marker>");
            if (routedColors.contains(color)) svg.append("<marker id=\"").append(id)
                    .append("-routed\" markerUnits=\"userSpaceOnUse\" markerWidth=\"8\" markerHeight=\"6\" refX=\"7\" refY=\"3\" orient=\"auto\"><path d=\"M0,0 L8,3 L0,6 z\" fill=\"")
                    .append(color).append("\"/></marker>");
        });
        svg.append("</defs>\n<rect width=\"100%\" height=\"100%\" fill=\"")
                .append(theme.canvasBackground()).append("\"/>\n");
        for (int line = 0; line < titleLines.size(); line++)
            text(svg, 14, 28 + line * 24, "start", 18, true, theme.canvasText("#0f172a"), titleLines.get(line));
        for (int line = 0; line < headerLinesText.size(); line++)
            text(svg, 14, 28 + titleLines.size() * 24 + line * 18, "start", 12, false,
                    theme.canvasText("#334155"), headerLinesText.get(line));
        // Keep all node/edge coordinates and routing unchanged inside one translated graph group.
        svg.append("<g class=\"architecture-graph\" transform=\"translate(0,").append(number(graphOffset)).append(")\">\n");
        var captionFont = font(11, false);
        var captionContext = new FontRenderContext(null, true, true);
        var metrics = captionFont.getLineMetrics("Ag", captionContext);
        var captions = new ArrayList<Rectangle2D>();
        for (var edge : scene.edges()) {
            double strokeWidth = 1.2 + Math.max(0, edge.relevance()) * 1.8;
            String status = canonical == null ? DiagramPresentation.relationshipStatus(edge.metadata(), language.german())
                    : DiagramPresentation.canonicalRelationshipStatus(canonical, edge.id(), language);
            var details = canonical == null ? DiagramPresentation.relationshipDetails(edge.metadata(), language.german())
                    : DiagramPresentation.canonicalRelationshipDetails(canonical, edge.id(), language);
            boolean proposed = proposed(edge, canonical);
            String stroke = theme.relationshipStroke(edge.relationType(), proposed);
            var detour = detours.get(edge);
            var canonicalRelation = canonical == null ? null : canonical.relationships().get(edge.id());
            String origin = canonical != null
                    ? canonicalRelation == null ? null : canonicalRelation.provenance()
                    : edge.metadata() == null || edge.metadata().origin() == null ? null : edge.metadata().origin().name();
            svg.append("<a xlink:href=\"#").append(edgeReferences.get(edge.id())).append("\"><g class=\"architecture-edge\" data-edge-id=\"").append(xml(edge.id()))
                .append("\" data-origin=\"").append(xml(origin == null || origin.isBlank() ? "unrecorded" : origin))
                .append("\"><title>").append(xml(edge.id() + " | " + edge.sourceId() + " -> " + edge.targetId() + " | " + edge.relationType()
                        + "\n" + String.join("\n", details))).append("</title>\n");
            if (detour == null) {
                svg.append("<line x1=\"").append(number(edge.sourceX())).append("\" y1=\"").append(number(edge.sourceY()))
                        .append("\" x2=\"").append(number(edge.targetX())).append("\" y2=\"").append(number(edge.targetY()));
            } else {
                svg.append("<polyline fill=\"none\" points=\"").append(detour.points().stream()
                        .map(point -> number(point.x()) + "," + number(point.y())).collect(Collectors.joining(" ")));
            }
            svg.append("\" stroke=\"").append(stroke).append("\" stroke-width=\"")
                .append(number(strokeWidth)).append('"');
            if (proposed) svg.append(" stroke-dasharray=\"6 4\"");
            svg.append(" marker-end=\"url(#").append(markers.get(stroke))
                    .append(detour == null ? "" : "-routed").append(")\"/>\n");
            svg.append("<metadata><relationship xmlns=\"urn:taxonomy:diagram\">");
            metadataValue(svg, "provenance", canonical == null ? edge.metadata() : canonical.relationships().get(edge.id()));
            svg.append("</relationship></metadata>\n");
            var anchors = DiagramEdgeRouting.captionAnchors(edge, detour);
            if (detour == null) {
                // SVG markers scale with the stroke. Keep the text clear of their full height.
                double markerClearance = 4 * strokeWidth + 2;
                double above = -Math.max(15, markerClearance + metrics.getDescent());
                double below = Math.max(23, markerClearance + metrics.getAscent());
                String kind = edge.relationType() == null || edge.relationType().isBlank()
                        ? language.text("Relationship", "Beziehung") : edge.relationType();
                drawCaption(svg, scene, anchors, kind, captionFont, captionContext, theme, captions, above, below);
                drawCaption(svg, scene, anchors, status, captionFont, captionContext, theme, captions, below, above);
            } else {
                if (!drawCaption(svg, scene, anchors, edge.relationType() + " · " + status,
                        captionFont, captionContext, theme, captions, -6, 16, -18, 28)) {
                    drawCaption(svg, scene, anchors, edge.relationType(), captionFont, captionContext,
                            theme, captions, -6, 16, -18, 28);
                    drawCaption(svg, scene, anchors, status, captionFont, captionContext, theme, captions, 16, -6, 28, -18);
                }
            }
            svg.append("</g></a>\n");
        }
        for (var node : scene.nodes()) {
            double nodeTextWidth = node.width() - 28;
            var assignment = node.responsibility();
            String ownerDetails = assignment == null ? "" : "\n" + String.join("\n", DiagramPresentation.responsibilityDetails(assignment, language));
            svg.append("<a xlink:href=\"#").append(nodeReferences.get(node.id())).append("\"><g class=\"architecture-node\" data-node-id=\"").append(xml(node.id()))
                .append("\" transform=\"translate(").append(number(node.x())).append(',').append(number(node.y())).append(")\">")
                .append("<title>").append(xml(node.id() + " | " + node.label() + " | " + node.type()
                        + (canonical == null ? " | " + language.depth(node) : "")
                        + (node.container() ? " | Container" : "")
                        + ownerDetails)).append("</title>\n")
                .append("<rect width=\"").append(number(node.width())).append("\" height=\"").append(number(node.height()))
                .append("\" rx=\"10\" fill=\"").append(theme.nodeFill(node.type())).append("\" stroke=\"")
                .append(theme.nodeBorder(node.type(), node.anchor())).append("\" stroke-width=\"").append(node.anchor() ? "3" : "1.3").append("\"/>\n");
            text(svg, 14, 17, "start", 10, true, theme.nodeText(node.type(), "#334155"), DiagramTextWrapper.wrapMeasured(node.id(), font(10, true), nodeTextWidth, 1, language.text("ID unrecorded", "Kennung nicht aufgezeichnet")).getFirst());
            int labelLines = assignment == null ? 2 : 1;
            if (canonical != null && node.height() >= LayeredDiagramLayoutService.NODE_HEIGHT) labelLines++;
            var labels = DiagramTextWrapper.wrapMeasured(node.label(), font(13, false), nodeTextWidth, labelLines,
                    language.text("Unnamed architecture element", "Unbenanntes Architekturelement"));
            double labelTop = canonical == null ? 36 : 30;
            double labelLeading = canonical == null ? 17 : 13;
            for (int i = 0; i < labels.size(); i++) text(svg, 14, labelTop + i * labelLeading,
                    "start", 13, false, theme.nodeText(node.type(), "#0f172a"), labels.get(i));
            if (assignment != null) {
                svg.append("<circle cx=\"18\" cy=\"52\" r=\"4\" fill=\"").append(assignment.color()).append("\"/>\n");
                text(svg, 26, 56, "start", 11, false, theme.nodeText(node.type(), "#0f172a"),
                        DiagramTextWrapper.wrapMeasured(assignment.name(), font(11, false), nodeTextWidth - 12, 1, "").getFirst());
            }
            text(svg, 14, 72, "start", 10, false, theme.nodeText(node.type(), "#475569"), DiagramTextWrapper.wrapMeasured(
                    canonical == null ? node.type() + " · " + language.score(node) : node.type(), font(10, false), nodeTextWidth, 1, language.text("Unclassified", "Nicht klassifiziert")).getFirst());
            svg.append("</g></a>\n");
        }
        svg.append("</g>\n");
        for (int i = 0; i < summary.size(); i++) text(svg, 14, footer + 16 + i * 16, "start", 11, false, theme.canvasText("#334155"),
                summary.get(i));
        if (!warningLines.isEmpty()) {
            text(svg, 14, warningTop, "start", 16, true, theme.canvasText("#0f172a"),
                    language.text("Recorded warnings and open work", "Gespeicherte Hinweise und offene Arbeit"));
            for (int line = 0; line < warningLines.size(); line++) text(svg, 14,
                    warningTop + 24 + line * EVIDENCE_LINE_HEIGHT, "start", EVIDENCE_FONT_SIZE, false,
                    theme.canvasText("#334155"), warningLines.get(line));
        }
        text(svg, 14, evidenceTitleY, "start", 16, true, theme.canvasText("#0f172a"), canonical == null ? language.text("Scores and complete explanations", "Bewertungen und vollständige Begründungen") : language.text("Canonical inventory", "Kanonisches Inventar"));
        for (int column = 0; column < columns; column++) {
            double x = 14 + column * EVIDENCE_COLUMN_WIDTH;
            var header = columnHeaders.get(column);
            for (int line = 0; line < header.size(); line++) text(svg, x,
                    evidenceTitleY + 28 + line * EVIDENCE_LINE_HEIGHT, "start", EVIDENCE_FONT_SIZE, true, theme.canvasText("#334155"), header.get(line));
            int index = column * linesPerColumn;
            int end = Math.min(evidence.size(), index + linesPerColumn);
            while (index < end) {
                var first = evidence.get(index);
                svg.append("<g class=\"architecture-evidence\" data-evidence-kind=\"").append(first.kind())
                        .append("\" data-evidence-id=\"").append(xml(first.id())).append('"');
                if (first.first()) svg.append(" id=\"").append(first.reference()).append('"');
                svg.append(">\n");
                do {
                    var line = evidence.get(index);
                    if (!line.value().isEmpty()) text(svg, x, evidenceTop + (index % linesPerColumn) * EVIDENCE_LINE_HEIGHT,
                            "start", EVIDENCE_FONT_SIZE, line.bold(), theme.canvasText("#0f172a"), line.value());
                    index++;
                } while (index < end && evidence.get(index).reference().equals(first.reference()));
                svg.append("</g>\n");
            }
        }
        return svg.append("</svg>\n").toString();
    }

    private static boolean drawCaption(StringBuilder svg, DiagramScene scene, List<DiagramEdgeRouting.Point> anchors,
            String label, Font font, FontRenderContext context, ExportTheme theme, List<Rectangle2D> occupied, double... offsets) {
        String caption = DiagramTextWrapper.wrapMeasured(label, font, 210, 1, "").getFirst();
        var glyphs = font.getStringBounds(caption, context);
        for (var anchor : anchors) {
            for (double offset : offsets) {
                double baseline = anchor.y() + offset;
                var box = new Rectangle2D.Double(anchor.x() - glyphs.getWidth() / 2, baseline + glyphs.getY(),
                        glyphs.getWidth(), glyphs.getHeight());
                if (box.getMinX() < 0 || box.getMaxX() > scene.width() || box.getMinY() < 0 || box.getMaxY() > scene.height()
                        || occupied.stream().anyMatch(box::intersects)
                        || scene.nodes().stream().anyMatch(node -> box.intersects(node.x(), node.y(), node.width(), node.height())))
                    continue;
                text(svg, anchor.x(), baseline, "middle", 11, false, theme.canvasText("#394150"), caption);
                occupied.add(box);
                return true;
            }
        }
        // The hyperlink still points to the full relationship label and status in the inventory.
        return false;
    }

    private static boolean proposed(com.taxonomy.diagram.DiagramSceneEdge edge, DiagramCanonicalEvidence canonical) {
        return canonical == null ? edge.metadata() != null && edge.metadata().proposed()
                : DiagramPresentation.canonicalRelationshipProposed(canonical, edge.id());
    }

    private record EvidenceLine(String kind, String id, String reference, String value, boolean bold, boolean first) { }

    private static List<EvidenceLine> evidence(DiagramScene scene, Map<String, String> nodeReasons,
                                               Map<String, Map<String, String>> relationReasons,
                                               Map<String, String> nodeReferences, Map<String, String> edgeReferences,
                                               DiagramLanguage language, DiagramCanonicalEvidence canonical) {
        var lines = new ArrayList<EvidenceLine>();
        var seenNodes = new HashSet<String>();
        for (var node : scene.nodes()) if (seenNodes.add(node.id())) {
            var parts = new ArrayList<String>();
            String metrics = canonical == null ? " | " + language.score(node) + " | " + language.depth(node) : "";
            parts.add(language.text("Node ", "Knoten ") + node.id() + metrics + (node.container() ? " | Container" : ""));
            parts.add(node.label());
            parts.add((node.type() == null ? language.text("Unclassified", "Nicht klassifiziert") : node.type()) + language.text(" | Parent: ", " | Übergeordnet: ")
                    + (node.parentId() == null ? language.text("not included", "nicht enthalten") : node.parentId()));
            parts.addAll(DiagramPresentation.responsibilityDetails(node.responsibility(), language));
            String reason = nodeReasons.get(node.id());
            if (reason != null && !reason.isBlank()) parts.add(language.text("Explanation: ", "Begründung: ") + reason);
            appendEvidence(lines, "node", node.id(), nodeReferences.get(node.id()), parts);
        }
        var seenEdges = new HashSet<String>();
        for (var edge : scene.edges()) if (seenEdges.add(edge.id())) {
            var parts = new ArrayList<String>();
            parts.add(language.text("Relationship ", "Beziehung ") + edge.id() + " | " + edge.sourceId() + " -> " + edge.targetId());
            if (canonical == null) {
                parts.add(edge.relationType() + " | " + language.percent(edge.relevance()) + " | " + edge.relationCategory());
                parts.addAll(DiagramPresentation.relationshipDetails(edge.metadata(), language.german()));
                parts.addAll(DiagramPresentation.relationshipEvidenceDetails(relationReasons.get(edge.id()), language.german()));
            } else {
                parts.add(edge.relationType());
                parts.addAll(DiagramPresentation.canonicalRelationshipDetails(canonical, edge.id(), language));
            }
            appendEvidence(lines, "relationship", edge.id(), edgeReferences.get(edge.id()), parts);
        }
        return lines;
    }

    private static void appendEvidence(List<EvidenceLine> lines, String kind, String id, String reference, List<String> parts) {
        boolean first = true;
        for (int part = 0; part < parts.size(); part++) {
            boolean bold = part == 0;
            for (String line : DiagramTextWrapper.wrapMeasured(parts.get(part), font(EVIDENCE_FONT_SIZE, bold),
                    EVIDENCE_TEXT_WIDTH, Integer.MAX_VALUE, "Unrecorded")) {
                lines.add(new EvidenceLine(kind, id, reference, line, bold, first));
                first = false;
            }
        }
        lines.add(new EvidenceLine(kind, id, reference, "", false, false));
    }

    private static Font font(int size, boolean bold) { return new Font("Arial", bold ? Font.BOLD : Font.PLAIN, size); }
    private static void text(StringBuilder svg, double x, double y, String anchor, int size, boolean bold, String color, String value) {
        svg.append("<text x=\"").append(number(x)).append("\" y=\"").append(number(y)).append("\" text-anchor=\"").append(anchor)
            .append("\" font-family=\"Arial, sans-serif\" font-size=\"").append(size).append('"');
        if (bold) svg.append(" font-weight=\"bold\"");
        if (!value.isEmpty()) svg.append(" textLength=\"").append(number(DiagramTextWrapper.measure(value, font(size, bold))))
            .append("\" lengthAdjust=\"spacingAndGlyphs\"");
        svg.append(" fill=\"").append(color).append("\">").append(xml(value)).append("</text>\n");
    }
    /** Record fields are separate XML elements, including explicit absence and nested scoped evidence. */
    private static void metadataValue(StringBuilder xml, String name, Object value) {
        xml.append('<').append(name);
        if (value == null) { xml.append(" recorded=\"false\"/>"); return; }
        xml.append('>');
        if (value instanceof java.util.List<?> items) {
            for (Object item : items) metadataValue(xml, "item", item);
        } else if (value.getClass().isRecord()) {
            for (var component : value.getClass().getRecordComponents()) {
                try { metadataValue(xml, component.getName(), component.getAccessor().invoke(value)); }
                catch (ReflectiveOperationException error) { throw new IllegalStateException("Cannot serialize recorded provenance", error); }
            }
        } else xml.append(xml(value instanceof Enum<?> e ? e.name() : value.toString()));
        xml.append("</").append(name).append('>');
    }
    private static String xml(String value) {
        return value == null ? "" : StringEscapeUtils.escapeXml10(value);
    }
    private static String number(double value) { return String.format(Locale.ROOT, "%.2f", value); }

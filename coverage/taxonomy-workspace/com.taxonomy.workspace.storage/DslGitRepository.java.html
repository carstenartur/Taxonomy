    /** Every newly constructed model commit must pass here before object insertion. */

    public CommitSignature verifyCommit(String commitId) throws IOException {
        return signatures.verify(rawCommit(commitId));
    }

    /** Validate a selected snapshot before copying its content to another repository. */
    public void verifySnapshot(String commitId) throws IOException {
        if (commitId == null) return;
        signatures.requireAcceptable(rawCommit(commitId));
        verifyProvenance(commitId);
    }

    public CommitProvenance getProvenance(String commitId) throws IOException {
        try (RevWalk walk = new RevWalk(gitRepo)) {
            var commit = walk.parseCommit(ObjectId.fromString(commitId));
            try (TreeWalk tree = TreeWalk.forPath(gitRepo, CommitProvenance.FILENAME, commit.getTree())) {
                if (tree == null) return null;
                if (!FileMode.REGULAR_FILE.equals(tree.getFileMode(0))) {
                    throw new IOException("Provenance must be a regular file");
                }
                return CommitProvenance.fromBytes(gitRepo.open(tree.getObjectId(0), Constants.OBJ_BLOB)
                        .getBytes(CommitProvenance.MAX_DOCUMENT_BYTES));
            }
        }
    }

    private void verifyProvenance(String commitId) throws IOException {
        var provenance = getProvenance(commitId);
        if (provenance != null) provenance.verify(signatures);
    }

    ObjectId insertDslTree(ObjectInserter inserter, String dslText, CommitProvenance provenance) throws IOException {
        TreeFormatter tree = new TreeFormatter();
        if (provenance != null) {
            provenance.verify(signatures);
            tree.append(CommitProvenance.FILENAME, FileMode.REGULAR_FILE,
                    inserter.insert(Constants.OBJ_BLOB, provenance.toBytes()));
        }
        tree.append(DSL_FILENAME, FileMode.REGULAR_FILE,
                inserter.insert(Constants.OBJ_BLOB, dslText.getBytes(StandardCharsets.UTF_8)));
        return inserter.insert(tree);
    }

    /** Check all newly introduced ancestry, including both sides of merges. */
    public void verifyIncoming(String head, String acceptedBase) throws IOException {
        verifyIncomingFromAcceptedHeads(head, acceptedBase == null ? List.of() : List.of(acceptedBase));
    }

    /** Check incoming history against local acceptance boundaries captured before a transport operation. */
    public void verifyIncomingFromAcceptedHeads(String head, Collection<String> acceptedHeads) throws IOException {
        if (head == null) return;
        Objects.requireNonNull(acceptedHeads, "acceptedHeads");
        try (RevWalk walk = new RevWalk(gitRepo)) {
            walk.markStart(walk.parseCommit(ObjectId.fromString(head)));
            for (String acceptedHead : acceptedHeads) {
                if (acceptedHead != null) {
                    walk.markUninteresting(walk.parseCommit(ObjectId.fromString(acceptedHead)));
                }
            }
            for (RevCommit commit : walk) {
                verifySnapshot(commit.name());
            }
        }
    }

    /** Capture accepted branches before publication; remote and temporary refs are not trust anchors. */
    public List<String> acceptedBranchHeads() throws IOException {
        return gitRepo.getRefDatabase().getRefsByPrefix(Constants.R_HEADS).stream()
                .filter(ref -> !ref.getName().startsWith(Constants.R_HEADS + EXTERNAL_SYNC_BRANCH_PREFIX))
                .filter(ref -> ref.getObjectId() != null)
                .map(ref -> ref.getObjectId().name())
                .distinct()
                .toList();
    }

    private byte[] rawCommit(String commitId) throws IOException {
        return gitRepo.open(ObjectId.fromString(commitId), Constants.OBJ_COMMIT)
                .getBytes(16 * 1024 * 1024);
    }

    /**
     * Commit DSL text to a branch and update the branch optimistically.

     * @return the new commit SHA
     * @throws IOException if object insertion or the ref update fails

    public String commitDsl(String branch, String dslText, String author, String message,
                            CommitProvenance provenance) throws IOException {
        PersonIdent actor = personIdent(author);
        String commitMessage = message != null ? message : "DSL update";
        String refName = Constants.R_HEADS + branch;
        Ref branchRef = gitRepo.getRefDatabase().exactRef(refName);

        try (ObjectInserter inserter = gitRepo.newObjectInserter()) {
            ObjectId treeId = insertDslTree(inserter, dslText, provenance);

            CommitBuilder commit = new CommitBuilder();
            commit.setTreeId(treeId);
            commit.setAuthor(actor);
            commit.setCommitter(actor);
            commit.setMessage(commitMessage);
            if (branchRef != null) {
                commit.setParentId(branchRef.getObjectId());
            }

            signCommit(commit, branch);
            ObjectId commitId = inserter.insert(commit);
            inserter.flush();

            RefUpdate.Result result = updateRef(
                    refName,
                    commitId,
                    branchRef != null ? branchRef.getObjectId() : ObjectId.zeroId(),
                    false,
                    actor,
                    "commit: " + commitMessage);

            log.info("Committed DSL to branch '{}': {} (ref update: {})",
                    branch, commitId.name(), result);
            return commitId.name();
        }
    }

    public boolean isAncestor(String ancestor, String descendant) throws IOException {
        if (ancestor == null || descendant == null) return false;
        try (RevWalk walk = new RevWalk(gitRepo)) {
            return walk.isMergedInto(walk.parseCommit(ObjectId.fromString(ancestor)), walk.parseCommit(ObjectId.fromString(descendant)));
        }
    }

    public String directBranchHead(String branch) throws IOException {
        Ref current = readFreshRef(Constants.R_HEADS + branch);
        if (current != null && current.isSymbolic()) throw new IOException("The selected branch is a symbolic reference");
        return current == null || current.getObjectId() == null ? null : current.getObjectId().name();
    }

    private String fastForwardIfHeadMatches(String branch, String expectedHead, String incomingHead, String author, String message,
                                            boolean directOnly) throws IOException {
        if (!Repository.isValidRefName(Constants.R_HEADS + branch)) throw new IllegalArgumentException("Invalid branch");
        // A ref-only publication must not overlap a transfer/checkpoint pack flush either.
        synchronized (gitRepo) {
            if (directOnly) directBranchHead(branch);
            verifyExpectedHead(branch, expectedHead);
            verifyIncoming(incomingHead, expectedHead);
            if (expectedHead != null && !isAncestor(expectedHead, incomingHead)) throw new IOException("Incoming history is not a descendant");
            RefUpdate update = gitRepo.updateRef(Constants.R_HEADS + branch, directOnly);
            update.setExpectedOldObjectId(expectedHead == null ? ObjectId.zeroId() : ObjectId.fromString(expectedHead));
            update.setNewObjectId(ObjectId.fromString(incomingHead)); update.setForceUpdate(false);
            update.setRefLogIdent(personIdent(author)); update.setRefLogMessage(message, true);
            RefUpdate.Result result = update.update();
            if (result == RefUpdate.Result.NEW || result == RefUpdate.Result.FAST_FORWARD || result == RefUpdate.Result.NO_CHANGE) return incomingHead;
            if (result == RefUpdate.Result.LOCK_FAILURE || result == RefUpdate.Result.REJECTED || result == RefUpdate.Result.REJECTED_CURRENT_BRANCH)
                throw new BranchHeadConflictException(branch, expectedHead, getHeadCommit(branch), "Branch changed during history integration");
            throw new IOException("History integration failed: " + result);
        }
    }

    /** Refresh independent storage handles before reading a remote observation or enforcing a Git precondition. */
    public Ref readFreshRef(String qualifiedName) throws IOException {
        return readFreshRef(gitRepo, qualifiedName);

    static Ref readFreshRef(Repository repository, String qualifiedName) throws IOException {
        synchronized (repository) {
            repository.scanForRepoChanges();
            return repository.exactRef(qualifiedName);
        }
    }

    /** Reject rewrites of an already fetched source before a pull or publish can modify either repository. */
    public void verifyUpstreamLineage(DslGitRepository source, String sourceRepositoryId, String branch, String observed)
            throws IOException {
        var upstream = new GitUpstreamRef(sourceRepositoryId, branch);
        String tracked = upstream.head(this);
        if (tracked == null) return;
        try {
            if (source.isAncestor(tracked, observed)) return;
        } catch (org.eclipse.jgit.errors.MissingObjectException
                 | org.eclipse.jgit.errors.IncorrectObjectTypeException missingHistory) {
            // A replaced source repository may no longer have the recorded commit.
        }
        throw new BranchHeadConflictException(branch, tracked, observed, "Source branch diverged from its last fetched commit");
    }

    /** Bind a newly provisioned target to its verified fork point, including an interrupted retry. */
    public void recordUpstreamForkPoint(String sourceRepositoryId, String sourceBranch, String commit, String actor) throws IOException {
        var upstream = new GitUpstreamRef(sourceRepositoryId, sourceBranch);
        String tracked = upstream.head(this);
        if (tracked != null && !tracked.equals(commit)) {
            throw new BranchHeadConflictException(
                    upstream.name(), commit, tracked, "Existing upstream ref differs from the recorded fork point");
        }
        upstream.advance(this, tracked, commit, actor);
    }

    /** A concurrent fetch may already have observed a descendant of this completed integration. */
    public String recordIntegratedUpstream(String sourceRepositoryId, String sourceBranch, String commit, String actor) throws IOException {
        var upstream = new GitUpstreamRef(sourceRepositoryId, sourceBranch);
        String tracked = upstream.head(this);
        if (tracked != null && isAncestor(commit, tracked)) return tracked;
        try {
            upstream.advance(this, tracked, commit, actor);
            return commit;
        } catch (BranchHeadConflictException changed) {
            String current = upstream.head(this);
            if (current != null && isAncestor(commit, current)) return current;
            throw changed;
        }
    }

    /** Bounded immutable-document read used by native history queries. */
    public String getDslAtCommit(String commitId, int maximumBytes) throws IOException {
        try (RevWalk walk = new RevWalk(gitRepo)) {
            var commit = walk.parseCommit(ObjectId.fromString(commitId));
            try (TreeWalk tree = TreeWalk.forPath(gitRepo, DSL_FILENAME, commit.getTree())) {
                if (tree == null) return "";
                ObjectLoader blob = gitRepo.open(tree.getObjectId(0), Constants.OBJ_BLOB);
                if (blob.getSize() > maximumBytes) throw new IOException("Historical DSL exceeds the document byte budget");
                return new String(blob.getBytes(maximumBytes), StandardCharsets.UTF_8);
            }
        }
    }

    /** Return commit history of a branch, newest first. */
    public List<DslCommit> getDslHistory(String branch) throws IOException {
        Ref ref = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + branch);
        if (ref == null) {
            return List.of();
        }

        List<DslCommit> history = new ArrayList<>();
        try (RevWalk walk = new RevWalk(gitRepo)) {
            walk.markStart(walk.parseCommit(ref.getObjectId()));
            for (RevCommit commit : walk) {
                PersonIdent authorIdent = commit.getAuthorIdent();
                history.add(new DslCommit(
                        commit.name(),
                        authorIdent.getName(),
                        Instant.ofEpochSecond(commit.getCommitTime()),
                        commit.getFullMessage(),
                        verifyCommit(commit.name())));
            }
        }
        return history;
    }

    // ── Branch operations ───────────────────────────────────────────

    /** List all local branches. */
    public List<DslBranch> listBranches() throws IOException {
        List<DslBranch> branches = new ArrayList<>();
        for (Ref ref : readFreshBranchRefs()) {
            String name = ref.getName().substring(Constants.R_HEADS.length());
            String headCommitId = ref.getObjectId() != null ? ref.getObjectId().name() : null;
            Instant created = oldestCommitTime(ref);
            branches.add(new DslBranch(name, headCommitId, created));
        }
        return branches;
    }

    private Instant oldestCommitTime(Ref ref) throws IOException {
        if (ref.getObjectId() == null) {
            return null;
        }
        try (RevWalk walk = new RevWalk(gitRepo)) {
            walk.markStart(walk.parseCommit(ref.getObjectId()));
            RevCommit oldest = null;
            for (RevCommit commit : walk) {
                oldest = commit;
            }
            return oldest != null ? Instant.ofEpochSecond(oldest.getCommitTime()) : null;
        }
    }

    /** Create a new branch at the HEAD of an existing branch. */
    public String createBranch(String newBranch, String fromBranch) throws IOException {
        Ref source = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + fromBranch);
        if (source == null) {
            return null;
        }

        String targetRef = Constants.R_HEADS + newBranch;
        updateRef(
                targetRef,
                source.getObjectId(),
                ObjectId.zeroId(),
                false,
                systemIdent(),
                "branch: created from " + fromBranch);

        log.info("Created branch '{}' from '{}' at {}",
                newBranch, fromBranch, source.getObjectId().name());
        return source.getObjectId().name();
    }

    /**
     * Create an absent branch at one captured commit, without re-reading another
     * branch's moving HEAD. Existing destination refs are never overwritten.
     */
    public String createBranchAtCommit(String newBranch, String commitId) throws IOException {
        ObjectId snapshot = ObjectId.fromString(commitId);
        try (RevWalk walk = new RevWalk(gitRepo)) {
            walk.parseCommit(snapshot);
        }
        updateRef(Constants.R_HEADS + newBranch, snapshot, ObjectId.zeroId(), false,
                systemIdent(), "branch: created at " + commitId);
        return snapshot.name();
    }

    /** Publish an explicitly selected snapshot only after checking all newly introduced ancestry. */
    public String createBranchAtVerifiedCommit(String newBranch, String commitId) throws IOException {
        List<String> accepted = acceptedBranchHeads();
        verifyIncomingFromAcceptedHeads(commitId, accepted);
        return createBranchAtCommit(newBranch, commitId);
    }

    // ── Diff operations ─────────────────────────────────────────────

    /** Compute a semantic model diff between two commits. */
    public ModelDiff diffBetween(String fromCommitId, String toCommitId) throws IOException {
        String beforeDsl = getDslAtCommit(fromCommitId);
        String afterDsl = getDslAtCommit(toCommitId);
        return differ.diff(toModel(beforeDsl), toModel(afterDsl));
    }

    /** Compute a semantic model diff between branch heads. */
    public ModelDiff diffBranches(String fromBranch, String toBranch) throws IOException {
        return differ.diff(toModel(getDslAtHead(fromBranch)), toModel(getDslAtHead(toBranch)));
    }

    private CanonicalArchitectureModel toModel(String dsl) {
        return dsl != null ? astMapper.map(parser.parse(dsl)) : null;
    }

    /** Produce a JGit-native unified text diff between two commits. */
    public String textDiff(String fromCommitId, String toCommitId) throws IOException {
        ObjectId fromObjectId = ObjectId.fromString(fromCommitId);
        ObjectId toObjectId = ObjectId.fromString(toCommitId);

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit fromCommit = walk.parseCommit(fromObjectId);
            RevCommit toCommit = walk.parseCommit(toObjectId);
            ByteArrayOutputStream output = new ByteArrayOutputStream();
            try (DiffFormatter formatter = new DiffFormatter(output)) {
                formatter.setRepository(gitRepo);
                formatter.format(fromCommit.getTree(), toCommit.getTree());
                formatter.flush();
            }
            return output.toString(StandardCharsets.UTF_8);
        }
    }

    /** Return JGit diff metadata between two commits. */
    public List<DiffEntry> jgitDiffEntries(String fromCommitId, String toCommitId)
            throws IOException {
        ObjectId fromObjectId = ObjectId.fromString(fromCommitId);
        ObjectId toObjectId = ObjectId.fromString(toCommitId);

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit fromCommit = walk.parseCommit(fromObjectId);
            RevCommit toCommit = walk.parseCommit(toObjectId);
            try (DiffFormatter formatter = new DiffFormatter(DisabledOutputStream.INSTANCE)) {
                formatter.setRepository(gitRepo);
                return formatter.scan(fromCommit.getTree(), toCommit.getTree());
            }
        }
    }

    // ── Cherry-pick and merge operations ────────────────────────────

    public record CherryPickTree(ObjectId resultTree, String baseCommit, List<String> conflictingPaths,
                                 boolean canResolveDsl) {}

    /** Preview the selected parent-to-commit change against an immutable target. */
    public CherryPickTree inspectCherryPick(String commitId, String targetCommit) throws IOException {
        try (var walk = new RevWalk(gitRepo)) {
            var picked = walk.parseCommit(ObjectId.fromString(commitId));
            var target = walk.parseCommit(ObjectId.fromString(targetCommit));
            return mergePickedChange(picked, target, null);
        }
    }

    private CherryPickTree mergePickedChange(RevCommit picked, RevCommit target, String resolvedDsl) throws IOException {
        if (picked.getParentCount() > 1) {
            throw new IllegalArgumentException("Cherry-picking a merge commit requires selecting a mainline parent");
        }
        var merger = new ModelHistoryMerger(gitRepo);
        String base = picked.getParentCount() == 0 ? null : picked.getParent(0).name();
        merger.useEmptyBase = base == null;
        if (base != null) merger.setBase(picked.getParent(0));
        merger.resolvedDsl = resolvedDsl;
        if (merger.merge(target, picked)) {
            return new CherryPickTree(merger.getResultTreeId(), base, List.of(), false);
        }
        var paths = new TreeSet<>(merger.getUnmergedPaths());
        if (merger.failed()) paths.addAll(merger.getFailingPaths().keySet());
        boolean dslOnly = !merger.failed() && paths.equals(Set.of(DSL_FILENAME))
                && editableDsl(target) && editableDsl(picked) && (base == null || editableDsl(picked.getParent(0)));
        return new CherryPickTree(null, base, List.copyOf(paths), dslOnly);
    }

    private boolean editableDsl(RevCommit commit) throws IOException {
        try (var walk = new RevWalk(gitRepo);
             var file = TreeWalk.forPath(gitRepo, DSL_FILENAME, walk.parseCommit(commit).getTree())) {
            if (file == null) return true;
            if (!FileMode.REGULAR_FILE.equals(file.getFileMode(0))) return false;
            try (var input = gitRepo.open(file.getObjectId(0), Constants.OBJ_BLOB).openStream()) {
                return !RawText.isBinary(input);
            }
        }
    }

    /** Prepare the exact default-branch change for an independently fenced adoption operation. */
    public DefaultCommitAdoption prepareDefaultCommit(String sourceRepository, String sourceBranch, String source,
                                                      String targetBranch, String target, String actor) throws IOException {
        verifyExpectedHead(targetBranch, target);
        try (var walk = new RevWalk(gitRepo)) {
            var picked = walk.parseCommit(ObjectId.fromString(source));
            var onto = walk.parseCommit(ObjectId.fromString(target));
            if (walk.isMergedInto(picked, onto))
                return new DefaultCommitAdoption("ALREADY_CONTAINED", null, "Source commit is in target history");
            if (picked.getParentCount() != 1)
                return new DefaultCommitAdoption("PREREQUISITE", null, "Select a mainline parent manually");
            verifyIncoming(source, target);
            var change = mergePickedChange(picked, onto, null);
            if (change.resultTree() == null)
                return new DefaultCommitAdoption("CONFLICT", null, String.join(", ", change.conflictingPaths()));
            if (DefaultCommitAdoption.sameContent(this, onto.getTree(), change.resultTree()))
                return new DefaultCommitAdoption("ALREADY_CONTAINED", null, "Target already contains this change");
            var provenance = CommitProvenance.capture("automatic-adoption", this, sourceRepository, sourceBranch,
                    source, change.baseCommit());
            String candidate = insertPickedCommit(targetBranch, picked, onto, change, provenance,
                    personIdent(actor), "Adopt " + sourceBranch + ": " + picked.getShortMessage());
            return new DefaultCommitAdoption("PREPARED", candidate, null);
        }
    }

    /** Apply a selected change at the current target head. */
    public String cherryPick(String commitId, String targetBranch) throws IOException {
        return cherryPick(commitId, targetBranch, getHeadCommit(targetBranch));
    }

    /** Apply the change only at the reviewed branch head. */
    public String cherryPick(String commitId, String targetBranch, String expectedTargetHead) throws IOException {
        return commitPickedChange(commitId, targetBranch, expectedTargetHead, null, null);
    }

    /** Resolve only a text conflict in the DSL while preserving every cleanly merged path. */
    public String resolveCherryPick(String commitId, String targetBranch, String expectedTargetHead,
                                    String resolvedDsl, String author) throws IOException {
        return commitPickedChange(commitId, targetBranch, Objects.requireNonNull(expectedTargetHead),
                Objects.requireNonNull(resolvedDsl), author);
    }

    private String commitPickedChange(String commitId, String targetBranch, String expectedTargetHead,
                                      String resolvedDsl, String author) throws IOException {
        verifyExpectedHead(targetBranch, expectedTargetHead);
        if (expectedTargetHead == null) return null;
        try (var walk = new RevWalk(gitRepo)) {
            var picked = walk.parseCommit(ObjectId.fromString(commitId));
            var target = walk.parseCommit(ObjectId.fromString(expectedTargetHead));
            verifyIncoming(commitId, expectedTargetHead);
            var change = mergePickedChange(picked, target, null);
            if (resolvedDsl != null) {
                if (!change.canResolveDsl()) return null;
                change = mergePickedChange(picked, target, resolvedDsl);
            }
            if (change.resultTree() == null) return null;

            String operation = resolvedDsl == null ? "cherry-pick" : "resolve-cherry-pick";
            var provenance = CommitProvenance.capture(operation, this, storageName, "detached", commitId, change.baseCommit());
            String message = (resolvedDsl == null ? "Cherry-pick: " : "Resolve cherry-pick: ") + picked.getShortMessage();
            PersonIdent committer = author == null ? systemIdent() : personIdent(author);
            String result = insertPickedCommit(targetBranch, picked, target, change, provenance, committer, message);
            return fastForwardIfHeadMatches(targetBranch, expectedTargetHead, result, committer.getName(), message);
        }
    }

    private String insertPickedCommit(String targetBranch, RevCommit picked, RevCommit target, CherryPickTree change,
                                      CommitProvenance provenance, PersonIdent committer, String message) throws IOException {
        try (var inserter = gitRepo.newObjectInserter()) {
            var commit = new CommitBuilder();
            commit.setTreeId(insertProvenanceTree(inserter, change.resultTree(), provenance));
            commit.setParentId(target);
            commit.setAuthor(picked.getAuthorIdent());
            commit.setCommitter(committer);
            commit.setMessage(message);
            signCommit(commit, targetBranch);
            ObjectId result = inserter.insert(commit);
            inserter.flush();
            return result.name();
        }
    }

    /**
     * Rewrite a reviewed, linear local history onto an already fetched upstream commit.
     *
     * <p>All merges and signed replacement commits are prepared as unreachable Git objects.
     * The selected branch is changed exactly once, with an expected-old-object CAS, only
     * after every commit has merged and the caller has revalidated current authority and
     * upstream refs. A conflict never partially rewrites the branch.</p>
     */
    @FunctionalInterface
    public interface RebaseGuard { void revalidate() throws IOException; }

    public record RebasedCommit(String originalCommit, String replacementCommit) { }

    public record RebaseReplay(String originalHead, String newHead, String conflictCommit,
                               List<String> conflictingPaths, List<RebasedCommit> rewritten, RebaseDslConflict dslConflict) {
        public boolean completed() { return conflictCommit == null; }
    }

    public RebaseReplay replayLinearHistory(String branch, String expectedHead, String mergeBase,
                                           String trackedCommit, List<String> orderedCommits,
                                           String actor, RebaseGuard reauthorizeAndCheckUpstream) throws IOException {
        return replayLinearHistory(branch, expectedHead, mergeBase, trackedCommit, orderedCommits, Map.of(), actor,
                reauthorizeAndCheckUpstream);
    }

    public RebaseReplay replayLinearHistory(String branch, String expectedHead, String mergeBase,
                                           String trackedCommit, List<String> orderedCommits,
                                           Map<String, RebaseDslConflict.Resolution> resolutions, String actor, RebaseGuard reauthorizeAndCheckUpstream) throws IOException {
        return replayLinearHistory(branch, expectedHead, mergeBase, trackedCommit, orderedCommits, resolutions, actor,
                reauthorizeAndCheckUpstream, true);
    }

    public RebaseReplay replayLinearHistory(String branch, String expectedHead, String mergeBase,
                                           String trackedCommit, List<String> orderedCommits,
                                           Map<String, RebaseDslConflict.Resolution> resolutions, String actor,
                                           RebaseGuard reauthorizeAndCheckUpstream, boolean publish) throws IOException {
        Objects.requireNonNull(orderedCommits, "orderedCommits");
        Objects.requireNonNull(reauthorizeAndCheckUpstream, "reauthorizeAndCheckUpstream");
        if (branch == null || !Repository.isValidRefName(Constants.R_HEADS + branch)
                || branch.startsWith(EXTERNAL_SYNC_BRANCH_PREFIX)
                || "main".equals(branch) || "accepted".equals(branch) || "sync-base".equals(branch)) {
            throw new IllegalArgumentException("An editable, exact local branch is required");
        }
        if (expectedHead == null || !ObjectId.isId(expectedHead)
                || mergeBase == null || !ObjectId.isId(mergeBase)
                || trackedCommit == null || !ObjectId.isId(trackedCommit)
                || orderedCommits.isEmpty() || orderedCommits.size() > 200
                || orderedCommits.stream().anyMatch(id -> id == null || !ObjectId.isId(id))) {
            throw new IllegalArgumentException("A complete, bounded and reviewed replay is required");
        }

        if (resolutions.size() > orderedCommits.size() || !orderedCommits.containsAll(resolutions.keySet())) {
            throw new IllegalArgumentException("Resolution outside the reviewed replay");
        }
        synchronized (gitRepo) {
            reauthorizeAndCheckUpstream.revalidate();
            verifyExpectedHead(branch, expectedHead);
            try (var walk = new RevWalk(gitRepo)) {
                var original = walk.parseCommit(ObjectId.fromString(expectedHead));
                var base = walk.parseCommit(ObjectId.fromString(mergeBase));
                var fetched = walk.parseCommit(ObjectId.fromString(trackedCommit));
                if (!walk.isMergedInto(base, original) || !walk.isMergedInto(base, fetched)
                        || walk.isMergedInto(original, fetched) || walk.isMergedInto(fetched, original)) {
                    throw new IllegalArgumentException("The reviewed branches no longer have a divergent linear history");
                }

                var originals = new ArrayList<RevCommit>();
                for (var cursor = original; !cursor.equals(base); ) {
                    if (cursor.getParentCount() != 1 || originals.size() >= 200) {
                        throw new IllegalArgumentException("Only bounded, linear local commits may be replayed");
                    }
                    originals.add(cursor);
                    cursor = walk.parseCommit(cursor.getParent(0));
                }
                java.util.Collections.reverse(originals);
                if (originals.size() != orderedCommits.size()) {
                    throw new IllegalArgumentException("The ordered replay no longer matches the reviewed plan");
                }
                for (int i = 0; i < originals.size(); i++) {
                    if (!originals.get(i).name().equals(orderedCommits.get(i))) {
                        throw new IllegalArgumentException("The ordered replay no longer matches the reviewed plan");
                    }
                }

                verifyIncoming(trackedCommit, expectedHead);
                var replacements = new ArrayList<RebasedCommit>();
                var tip = fetched;
                for (var picked : originals) {
                    var change = mergePickedChange(picked, tip, null);
                    var resolution = resolutions.get(picked.name());
                    RebaseDslConflict conflict = null;
                    if (change.canResolveDsl()) {
                        conflict = RebaseDslConflict.read(gitRepo, walk.parseCommit(picked.getParent(0)), tip, picked);
                    }
                    if (resolution != null) {
                        if (conflict == null || !resolution.targetBlobId().equals(conflict.target().blobId())) {
                            throw new BranchHeadConflictException(branch, expectedHead, original.name(),
                                    "The reviewed model conflict changed");
                        }
                        change = mergePickedChange(picked, tip, resolution.text());
                    }
                    if (change.resultTree() == null) {
                        reauthorizeAndCheckUpstream.revalidate();
                        verifyExpectedHead(branch, expectedHead);
                        return new RebaseReplay(expectedHead, expectedHead, picked.name(),
                                change.conflictingPaths(), List.copyOf(replacements), conflict);
                    }
                    var evidence = CommitProvenance.capture(resolution == null ? "rebase" : "resolve-rebase", this, storageName, branch,
                            picked.name(), picked.getParent(0).name());
                    String summary = picked.getShortMessage();
                    if (summary.length() > 160) summary = summary.substring(0, 160);
                    String message = "Rebase: " + summary + "\n\nOriginal-Commit: " + picked.name();
                    try (var inserter = gitRepo.newObjectInserter()) {
                        var commit = new CommitBuilder();
                        commit.setTreeId(insertProvenanceTree(inserter, change.resultTree(), evidence));
                        commit.setParentId(tip);
                        commit.setAuthor(picked.getAuthorIdent());
                        commit.setCommitter(personIdent(actor));
                        commit.setMessage(message);
                        signCommit(commit, branch);
                        var id = inserter.insert(commit);
                        inserter.flush();
                        tip = walk.parseCommit(id);
                        replacements.add(new RebasedCommit(picked.name(), id.name()));
                    }
                }

                // No user branch has moved so far. Check both authority and all upstream refs
                // again before the single forced (non-fast-forward) compare-and-set publication.
                reauthorizeAndCheckUpstream.revalidate();
                verifyExpectedHead(branch, expectedHead);
                if (publish) ExpectedHeadDslCommitter.replaceBranchHead(gitRepo, branch, original, tip,
                        personIdent(actor), "rebase: " + original.name() + " onto " + fetched.name());
                return new RebaseReplay(expectedHead, tip.name(), null, List.of(), List.copyOf(replacements), null);
            }
        }
    }

        return merge(fromBranch, intoBranch, null, null);
    }

    /** Supply final merge metadata before signing and publishing the single new object. */
    public String merge(String fromBranch, String intoBranch, String author, String message) throws IOException {
        return merge(fromBranch, intoBranch, author, message, null, ignored -> { });
    }

    /** Check the exact candidate DSL before the captured merge can publish a ref. */
    public String merge(String fromBranch, String intoBranch, String author, String message,
                        Consumer<String> validateMergedSource) throws IOException {
        return merge(fromBranch, intoBranch, author, message, null, Objects.requireNonNull(validateMergedSource));
    }

    /** Merge transport history using only the locally accepted heads captured before fetching. */
    public String mergeFromAcceptedHeads(String fromBranch, String intoBranch,
                                         Collection<String> acceptedHeads) throws IOException {
        return merge(fromBranch, intoBranch, null, null, List.copyOf(acceptedHeads), ignored -> { });
    }

    private String merge(String fromBranch, String intoBranch, String author, String message,
                         Collection<String> acceptedHeads, Consumer<String> validateMergedSource) throws IOException {
        String fromRefName = Constants.R_HEADS + fromBranch;
        String intoRefName = Constants.R_HEADS + intoBranch;
        Ref fromRef = gitRepo.getRefDatabase().exactRef(fromRefName);
        Ref intoRef = gitRepo.getRefDatabase().exactRef(intoRefName);
        if (fromRef == null || intoRef == null) {
            log.warn("Cannot merge: branch not found (from={}, into={})", fromBranch, intoBranch);
            return null;
        }

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit fromCommit = walk.parseCommit(fromRef.getObjectId());
            RevCommit intoCommit = walk.parseCommit(intoRef.getObjectId());
            PersonIdent actor = author == null || author.isBlank() ? systemIdent() : personIdent(author);

            if (acceptedHeads == null) {
                verifyIncoming(fromCommit.name(), intoCommit.name());
            } else {
                verifyIncomingFromAcceptedHeads(fromCommit.name(), acceptedHeads);
            }

            if (walk.isMergedInto(fromCommit, intoCommit)) {
                validateMergeSource(validateMergedSource, intoCommit.getTree(), fromBranch, fromCommit, intoBranch, intoCommit);
                log.info("Already merged: '{}' is ancestor of '{}'", fromBranch, intoBranch);
                return intoCommit.name();
            }

            if (walk.isMergedInto(intoCommit, fromCommit)) {
                validateMergeSource(validateMergedSource, fromCommit.getTree(), fromBranch, fromCommit, intoBranch, intoCommit);
                updateRef(
                        intoRefName,
                        fromCommit,
                        intoCommit,
                        false,
                        actor,
                        "merge: fast-forward " + fromBranch + " into " + intoBranch);
                log.info("Fast-forward merge '{}' into '{}': {}",
                        fromBranch, intoBranch, fromCommit.name());
                return fromCommit.name();
            }

            ThreeWayMerger merger = new ModelHistoryMerger(gitRepo);
            if (!merger.merge(intoCommit, fromCommit)) {
                log.warn("Merge conflict: '{}' into '{}'", fromBranch, intoBranch);
                return null;
            }

            validateMergeSource(validateMergedSource, walk.parseTree(merger.getResultTreeId()),
                    fromBranch, fromCommit, intoBranch, intoCommit);

            try (ObjectInserter inserter = gitRepo.newObjectInserter()) {
                CommitBuilder mergeCommit = new CommitBuilder();
                mergeCommit.setTreeId(merger.getResultTreeId());
                mergeCommit.setParentIds(intoCommit, fromCommit);
                mergeCommit.setAuthor(actor);
                mergeCommit.setCommitter(actor);
                mergeCommit.setMessage(message == null ? "Merge branch '" + fromBranch + "' into " + intoBranch : message);

                signCommit(mergeCommit, intoBranch);
                ObjectId mergeCommitId = inserter.insert(mergeCommit);
                inserter.flush();
                updateRef(
                        intoRefName,
                        mergeCommitId,
                        intoCommit,
                        false,
                        actor,
                        "merge: " + fromBranch + " into " + intoBranch);

                log.info("Merged '{}' into '{}': {}",
                        fromBranch, intoBranch, mergeCommitId.name());
                return mergeCommitId.name();
            }
        }
    }

    private void validateMergeSource(Consumer<String> validation, RevTree tree, String fromBranch,
                                     RevCommit fromCommit, String intoBranch, RevCommit intoCommit) throws IOException {
        String dsl = readDslFromTree(tree);
        // Application preflight may read its database; never hold the Repository monitor here.
        validation.accept(dsl == null ? "" : dsl);
        verifyExpectedHead(fromBranch, fromCommit.name());
        verifyExpectedHead(intoBranch, intoCommit.name());
    }

    /** Model-content merges exclude the reserved, transfer-specific metadata path. */
    private static final class ModelHistoryMerger extends RecursiveMerger {
        private boolean useEmptyBase;
        private String resolvedDsl;

        @Override
        protected AbstractTreeIterator mergeBase() throws IOException {
            return useEmptyBase ? new EmptyTreeIterator() : super.mergeBase();
        }

        ModelHistoryMerger(Repository repository) {
            super(repository, true);
        }

        @Override
        protected boolean processEntry(CanonicalTreeParser base, CanonicalTreeParser ours,
                                       CanonicalTreeParser theirs, DirCacheBuildIterator index,
                                       WorkingTreeIterator work, boolean ignoreConflicts,
                                       MergeAttributes attributes) throws IOException {
            if (CommitProvenance.FILENAME.equals(tw.getPathString())) {
                // Do not add the reserved transfer-specific path to the in-core
                // index. All other paths keep JGit's recursive merge behavior.
                enterSubtree = false;
                return true;
            }
            if (resolvedDsl != null && DSL_FILENAME.equals(tw.getPathString())) {
                byte[] content = resolvedDsl.getBytes(StandardCharsets.UTF_8);
                ObjectId blob = getObjectInserter().insert(Constants.OBJ_BLOB, content);
                workTreeUpdater.addExistingToIndex(blob, tw.getRawPath(), FileMode.REGULAR_FILE,
                        DirCacheEntry.STAGE_0, null, content.length);
                enterSubtree = false;
                return true;
            }
            return super.processEntry(base, ours, theirs, index, work, ignoreConflicts, attributes);
        }
    }

    /** Single-parent operations need fresh source evidence while retaining every other merged path. */
    private ObjectId insertProvenanceTree(ObjectInserter inserter, ObjectId mergedTree,
                                         CommitProvenance provenance) throws IOException {
        provenance.verify(signatures);
        ObjectId proof = inserter.insert(Constants.OBJ_BLOB, provenance.toBytes());
        try (var reader = gitRepo.newObjectReader()) {
            DirCache index = DirCache.read(reader, mergedTree);
            DirCacheEditor editor = index.editor();
            editor.add(new DirCacheEditor.PathEdit(CommitProvenance.FILENAME) {
                @Override
                public void apply(DirCacheEntry entry) {
                    entry.setFileMode(FileMode.REGULAR_FILE);
                    entry.setObjectId(proof);
                }
            });
            editor.finish();
            return index.writeTree(inserter);
        }
    }

    // ── Revert and undo operations ──────────────────────────────────

    /** Revert a specific commit on a branch by creating a new commit. */
    public String revert(String commitId, String branch) throws IOException {
        ObjectId revertObjectId = ObjectId.fromString(commitId);
        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit revertCommit = walk.parseCommit(revertObjectId);
            verifySnapshot(commitId);
            if (revertCommit.getParentCount() == 0) {
                log.warn("Cannot revert initial commit {}", commitId);
                return null;
            }

            String refName = Constants.R_HEADS + branch;
            Ref branchRef = gitRepo.getRefDatabase().exactRef(refName);
            if (branchRef == null) {
                log.warn("Branch '{}' not found for revert", branch);
                return null;
            }

            RevCommit branchHead = walk.parseCommit(branchRef.getObjectId());
            RevCommit parentCommit = walk.parseCommit(revertCommit.getParent(0));
            verifySnapshot(parentCommit.name());
            ThreeWayMerger merger = new ModelHistoryMerger(gitRepo);
            merger.setBase(revertCommit);
            if (!merger.merge(branchHead, parentCommit)) {
                log.warn("Revert conflict for {} on '{}'", commitId, branch);
                return null;
            }

            PersonIdent actor = systemIdent();
            var provenance = CommitProvenance.capture("revert", this, storageName,
                    "detached", parentCommit.name(), commitId);
            try (ObjectInserter inserter = gitRepo.newObjectInserter()) {
                CommitBuilder newCommit = new CommitBuilder();
                newCommit.setTreeId(insertProvenanceTree(inserter, merger.getResultTreeId(), provenance));
                newCommit.setParentId(branchHead);
                newCommit.setAuthor(actor);
                newCommit.setCommitter(actor);
                newCommit.setMessage("Revert: " + revertCommit.getShortMessage());

                signCommit(newCommit, branch);
                ObjectId newCommitId = inserter.insert(newCommit);
                inserter.flush();
                updateRef(
                        refName,
                        newCommitId,
                        branchHead,
                        false,
                        actor,
                        "revert: " + revertCommit.getShortMessage());

                log.info("Reverted {} on '{}': {}", commitId, branch, newCommitId.name());
                return newCommitId.name();
            }
        }
    }

    /** Move a branch back to its first parent. */
    public String undoLast(String branch) throws IOException {
        String refName = Constants.R_HEADS + branch;
        Ref branchRef = gitRepo.getRefDatabase().exactRef(refName);
        if (branchRef == null) {
            log.warn("Branch '{}' not found for undo", branch);
            return null;
        }

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit headCommit = walk.parseCommit(branchRef.getObjectId());
            if (headCommit.getParentCount() == 0) {
                log.warn("Cannot undo: branch '{}' has only the initial commit", branch);
                return null;
            }

            RevCommit parentCommit = walk.parseCommit(headCommit.getParent(0));
            updateRef(
                    refName,
                    parentCommit,
                    headCommit,
                    true,
                    systemIdent(),
                    "reset: moving to " + parentCommit.name());

            log.info("Undo last on '{}': {} -> {}",
                    branch, headCommit.name(), parentCommit.name());
            return parentCommit.name();
        }
    }

    /** Restore DSL content from a historic commit as a new branch commit. */
    public String restore(String commitId, String branch) throws IOException {
        String dslText;
        try {
            dslText = getDslAtCommit(commitId);
        } catch (org.eclipse.jgit.errors.MissingObjectException exception) {
            log.warn("Cannot restore: commit {} not found", commitId);
            return null;
        }
        if (dslText == null) {
            log.warn("Cannot restore: commit {} has no DSL content", commitId);
            return null;
        }
        var provenance = CommitProvenance.capture("restore", this, storageName,
                "detached", commitId, null);
        return commitDsl(
                branch,
                dslText,
                "taxonomy",
                "Restored from version " + commitId.substring(0, 7), provenance);
    }

    // ── State query helpers ─────────────────────────────────────────

    /** Read the current branch head, including updates published through another repository handle. */
    public String getHeadCommit(String branch) throws IOException {
        Ref ref = readFreshRef(Constants.R_HEADS + branch);
        return ref != null && ref.getObjectId() != null ? ref.getObjectId().name() : null;
    }

    /** Return all branch names without the {@code refs/heads/} prefix. */
    public List<String> getBranchNames() throws IOException {
        List<String> names = new ArrayList<>();
        for (Ref ref : readFreshBranchRefs()) {
            names.add(ref.getName().substring(Constants.R_HEADS.length()));
        }
        return names;
    }

    /** Include ref changes made through independent handles, such as reviewed variant deletion. */
    private List<Ref> readFreshBranchRefs() throws IOException {
        synchronized (gitRepo) {
            gitRepo.scanForRepoChanges();
            return gitRepo.getRefDatabase().getRefsByPrefix(Constants.R_HEADS);
        }
    }

    /** Count commits reachable from a branch. */
    public int getCommitCount(String branch) throws IOException {
        Ref ref = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + branch);
        if (ref == null) {
            return 0;
        }

        int count = 0;
        try (RevWalk walk = new RevWalk(gitRepo)) {
            walk.markStart(walk.parseCommit(ref.getObjectId()));
            for (RevCommit ignored : walk) {
                count++;
            }
        }
        return count;
    }

    /** Return {@code [ahead, behind]} commit counts for two branches. */
    public int[] getAheadBehindCounts(String branch, String baseBranch) throws IOException {
        Ref branchRef = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + branch);
        Ref baseRef = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + baseBranch);
        if (branchRef == null || baseRef == null) {
            return new int[]{0, 0};
        }

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit branchHead = walk.parseCommit(branchRef.getObjectId());
            RevCommit baseHead = walk.parseCommit(baseRef.getObjectId());
            if (branchHead.equals(baseHead)) {
                return new int[]{0, 0};
            }

            int ahead = countCommitsNotIn(walk, branchHead, baseHead);
            int behind = countCommitsNotIn(walk, baseHead, branchHead);
            return new int[]{ahead, behind};
        }
    }

    /** Compare selected branches in separate object stores without importing objects or changing refs. */
    public int[] getAheadBehindCounts(String branch, DslGitRepository source, String sourceBranch) throws IOException {
        if (source == this) return getAheadBehindCounts(branch, sourceBranch);
        String localHead = getHeadCommit(branch);
        String sourceHead = source.getHeadCommit(sourceBranch);
        if (Objects.equals(localHead, sourceHead)) return new int[]{0, 0};

        var localOnly = new java.util.HashSet<ObjectId>();
        try (var local = new RevWalk(gitRepo); var upstream = new RevWalk(source.gitRepo)) {
            if (localHead != null) {
                local.markStart(local.parseCommit(ObjectId.fromString(localHead)));
                for (var commit : local) localOnly.add(commit.copy());
            }
            int behind = 0;
            if (sourceHead != null) {
                upstream.markStart(upstream.parseCommit(ObjectId.fromString(sourceHead)));
                for (var commit : upstream) {
                    if (!localOnly.remove(commit)) behind++;
                }
            }
            return new int[]{localOnly.size(), behind};
        }
    }

    private static int countCommitsNotIn(
            RevWalk walk,
            RevCommit tip,
            RevCommit exclude) throws IOException {
        walk.reset();
        walk.markStart(tip);
        walk.markUninteresting(exclude);
        int count = 0;
        for (RevCommit ignored : walk) {
            count++;
        }
        return count;
    }

    /** Return metadata for the HEAD commit of a branch. */
    public DslCommit getHeadCommitInfo(String branch) throws IOException {
        Ref ref = gitRepo.getRefDatabase().exactRef(Constants.R_HEADS + branch);
        if (ref == null) {
            return null;
        }

        try (RevWalk walk = new RevWalk(gitRepo)) {
            RevCommit commit = walk.parseCommit(ref.getObjectId());
            PersonIdent authorIdent = commit.getAuthorIdent();
            return new DslCommit(
                    commit.name(),
                    authorIdent.getName(),
                    Instant.ofEpochSecond(commit.getCommitTime()),
                    commit.getFullMessage(),
                    verifyCommit(commit.name()));
        }
    }

    private String readDslFromTree(RevTree tree) throws IOException {
        try (TreeWalk treeWalk = TreeWalk.forPath(gitRepo, DSL_FILENAME, tree)) {
            if (treeWalk == null) {
                return null;
            }
            ObjectLoader loader = gitRepo.open(treeWalk.getObjectId(0), Constants.OBJ_BLOB);
            return new String(loader.getBytes(), StandardCharsets.UTF_8);
        }
    }

    private RefUpdate.Result updateRef(
            String refName,
            ObjectId newObjectId,
            ObjectId expectedOldObjectId,
            boolean force,
            PersonIdent actor,
            String reflogMessage) throws IOException {
        RefUpdate update = gitRepo.updateRef(refName);
        update.setNewObjectId(newObjectId);
        update.setExpectedOldObjectId(expectedOldObjectId);
        update.setForceUpdate(force);
        update.setRefLogIdent(actor);
        update.setRefLogMessage(reflogMessage, true);
        RefUpdate.Result result;
        try {
            result = update.update();
        } catch (IOException failure) {
            throw unwrapRefUpdateFailure(failure);
        }
        requireSuccessfulUpdate(refName, result);
        return result;
    }

    /** Preserve a policy denial through JGit's storage exception wrapper. */
    static IOException unwrapRefUpdateFailure(IOException failure) {
        Throwable cause = org.springframework.core.NestedExceptionUtils.getMostSpecificCause(failure);
        if (cause instanceof org.springframework.security.access.AccessDeniedException denied) throw denied;
        return failure;
    }

    private static void requireSuccessfulUpdate(String refName, RefUpdate.Result result)
            throws IOException {
        if (result != RefUpdate.Result.NEW
                && result != RefUpdate.Result.FAST_FORWARD
                && result != RefUpdate.Result.FORCED
                && result != RefUpdate.Result.NO_CHANGE) {
            throw new IOException("Ref update failed for " + refName + ": " + result);
        }
    }

    private static PersonIdent personIdent(String author) {
        String name = author != null ? author : "taxonomy";
        String email = author != null ? author : "taxonomy@system";
        return new PersonIdent(name, email);
    }

    private static PersonIdent systemIdent() {
        return new PersonIdent("taxonomy", "taxonomy@system");
    }

    public Repository getGitRepository() {
        return gitRepo;
    }

    /** Return whether this instance uses persistent Hibernate-backed storage. */
    public boolean isDatabaseBacked() {
        return databaseBacked;

    /** These workflow refs are managed by sync/publication and cannot be removed as user variants. */
    public static boolean isProtectedVariant(String name) {
        return "draft".equals(name) || "accepted".equals(name) || "main".equals(name) || "sync-base".equals(name);
    }

    /** Delete a non-protected branch and record a queryable reflog entry. */
    public boolean deleteBranch(String branchName) throws IOException {
        return deleteBranch(branchName, null, null);
    }

    /** Delete the reviewed ref only if the stored head still matches its exact object ID. */
    public boolean deleteBranch(String branchName, String expectedHead, String actor) throws IOException {
        if (isProtectedVariant(branchName)) {
            throw new IllegalArgumentException("Cannot delete protected branch: " + branchName);
        }
        String refName = Constants.R_HEADS + branchName;
        if (branchName == null || !Repository.isValidRefName(refName)) throw new IllegalArgumentException("Invalid branch");
        if (expectedHead != null && !ObjectId.isId(expectedHead)) throw new IllegalArgumentException("Invalid expected commit");

        Ref ref = gitRepo.getRefDatabase().exactRef(refName);
        ObjectId actual = ref == null ? null : ref.getObjectId();
        if (expectedHead != null && (actual == null || !ObjectId.fromString(expectedHead).equals(actual))) {
            throw new BranchHeadConflictException(branchName, expectedHead, actual == null ? null : actual.name(),
                    "The reviewed branch changed");
        }
        if (actual == null) return false;
        if (ref.isSymbolic()) throw new IllegalArgumentException("Cannot delete a symbolic branch");

        RefUpdate update = gitRepo.updateRef(refName, true);
        update.setExpectedOldObjectId(actual);
        update.setForceUpdate(true);
        update.setRefLogIdent(actor == null ? systemIdent() : personIdent(actor));
        update.setRefLogMessage("branch: deleted " + branchName, true);
        RefUpdate.Result result;
        try {
            result = update.delete();
        } catch (IOException failure) {
            throw unwrapRefUpdateFailure(failure);
        }
        if (expectedHead != null && (result == RefUpdate.Result.LOCK_FAILURE || result == RefUpdate.Result.REJECTED
                || result == RefUpdate.Result.REJECTED_CURRENT_BRANCH || result == RefUpdate.Result.NEW)) {
            throw new BranchHeadConflictException(branchName, expectedHead, getHeadCommit(branchName),
                    "The reviewed branch could not be deleted at its expected head");
        }
        if (result != RefUpdate.Result.FORCED
                && result != RefUpdate.Result.FAST_FORWARD) {
            throw new IOException("Ref deletion failed for " + refName + ": " + result);
        }

        log.info("Deleted branch '{}' (ref update: {})", branchName, result);
        return true;
    }

    /** Close the library handle or an owned in-memory repository exactly once. */
    @Override
    public void close() {
        if (!closed.compareAndSet(false, true)) {
            return;
        }
        if (storageHandle != null) {
            storageHandle.close();
        } else if (securedHandle != null) {
            securedHandle.close();
        } else if (closeRepository) {
            gitRepo.close();
        }
    }

    @org.springframework.beans.factory.annotation.Autowired
    public ProjectRequirementAnalysisService(
            ProjectPortfolioService projectService,
            PortfolioAnalysisPersistenceService persistenceService,
            PortfolioAnalysisClaimPersistenceService claimPersistenceService,
            PortfolioAnalysisWorkQueue workQueue,
            PortfolioAnalysisRecoveryService recoveryService,
            AnalyzeRequirementUseCase analyzeRequirementUseCase,
            AnalysisCopilotService copilotService,
            PortfolioFingerprintService fingerprintService,
            LlmService llmService,
            @Qualifier("portfolioAnalysisExecutor")
            AsyncTaskExecutor analysisExecutor,
            @Value("${taxonomy.limits.max-architecture-nodes:100}")
            int maximumArchitectureNodes,
            @Value("${taxonomy.portfolio.max-analysis-batch:100}")
            int maximumBatchRequirements,
            @Value("${taxonomy.portfolio.analysis-claim-timeout-seconds:900}")
            long claimTimeoutSeconds) {
        this.projectService = projectService;
        this.persistenceService = persistenceService;
        this.claimPersistenceService = claimPersistenceService;
        this.workQueue = workQueue;
        this.recoveryService = recoveryService;
        this.analyzeRequirementUseCase = analyzeRequirementUseCase;
        this.copilotService = Objects.requireNonNull(copilotService);
        this.fingerprintService = fingerprintService;
        this.llmService = llmService;
        this.analysisExecutor = analysisExecutor;
        this.maximumArchitectureNodes = Math.max(1, maximumArchitectureNodes);
        this.maximumBatchRequirements = Math.max(1, maximumBatchRequirements);
        this.claimTimeoutSeconds = Math.max(60L, claimTimeoutSeconds);
    }

    /** Executes a persisted job in the caller thread. Kept for internal batch use and deterministic tests. */
    public AnalysisJobView analyzeProject(Long projectId,
                                          AnalyzeProjectRequest request,
                                          String username,
                                          WorkspaceContext context) {
        var owner = admitOwner(username, context);
        AnalysisJobView job = prepareJob(projectId, request, username, context);
        if (job.status() != AnalysisStatus.PENDING) {
            return job;
        }
        return executePendingItems(job.id(), projectId, username, context, owner);
    }

    /** Persists and dispatches a job, returning before any LLM request is made. */
    public AnalysisJobView enqueueProject(Long projectId,
                                          AnalyzeProjectRequest request,
                                          String username,
                                          WorkspaceContext context) {
        var owner = admitOwner(username, context);
        AnalysisJobView job = prepareJob(projectId, request, username, context);
        return dispatchPending(job, projectId, username, context, owner);
    }

    public AnalysisJobView enqueueVerificationPass(Long projectId, Long requirementId, CopilotOperationKey key,
                                                  String username, WorkspaceContext context) {
        var owner = admitOwner(username, context);
        var job = persistenceService.createOrReuseVerificationPass(projectId, requirementId, key, username, context);
        return dispatchPending(job, projectId, username, context, owner);
    }

    /** Recheck the original request's workspace and login before any scheduled pass or promotion. */
    void checkCopilotOwner(AnalysisOwner owner, String username, WorkspaceContext context) {
        if (owner != null) budgetContinuations.checkOwner(owner, username, context);
    }

    AnalysisJobView enqueueAdmittedPass(Long projectId, Long requirementId, CopilotOperationKey key,
            String provider, int maxNodes, String username, WorkspaceContext context, AnalysisOwner owner) {
        checkCopilotOwner(owner, username, context);
        var job = key.pass() == 1
                ? prepareJob(projectId, requestForRequirement(requirementId, provider, maxNodes, key.value()), username, context)
                : persistenceService.createOrReuseVerificationPass(projectId, requirementId, key, username, context);
        return dispatchPending(job, projectId, username, context, owner);
    }

    private AnalysisJobView dispatchPending(AnalysisJobView job, Long projectId, String username, WorkspaceContext context,
                                            AnalysisOwner owner) {
        if (job.status() == AnalysisStatus.PENDING) {
            dispatch(job.id(), projectId, username, context, owner);
        }
        return job;
    }

    public AnalysisJobView retryFailed(String jobId,
                                       Long projectId,
                                       String username,
                                       WorkspaceContext context) {
        var owner = admitOwner(username, context);
        recoveryService.prepareRetryableItems(
                jobId,
                projectId,
                username,
                context,
                Instant.now().minusSeconds(claimTimeoutSeconds));
        return executePendingItems(jobId, projectId, username, context, owner);
    }

    /** Re-dispatches pending jobs or recovers failed/expired claims without holding the HTTP request. */
    public AnalysisJobView enqueueRetryFailed(String jobId,
                                              Long projectId,
                                              String username,
                                              WorkspaceContext context) {
        var owner = admitOwner(username, context);
        AnalysisJobView job = persistenceService.getJob(jobId, projectId, username, context);
        if (job.status() == AnalysisStatus.PENDING) {
            requireJobProviderBinding(job, username, context);
        } else {
            recoveryService.prepareRetryableItems(
                    jobId,
                    projectId,
                    username,
                    context,
                    Instant.now().minusSeconds(claimTimeoutSeconds));
            job = persistenceService.getJob(jobId, projectId, username, context);
        }
        dispatch(jobId, projectId, username, context, owner);
        return job;
    }

    /** One explicit action authorizes one further ordinary, bounded worker attempt. */
    public AnalysisJobView enqueueContinuation(String jobId, Long projectId, Long itemId,
            ContinueAnalysisRequest request, String username, WorkspaceContext context) {
        var owner = admitOwner(username, context);
        requireContinuations().prepare(jobId, projectId, itemId, request, username, context);
        AnalysisJobView prepared = persistenceService.getJob(jobId, projectId, username, context);
        dispatch(jobId, projectId, username, context, owner);
        return prepared;
    }

    private AnalysisJobView prepareJob(Long projectId,
                                       AnalyzeProjectRequest request,
                                       String username,
                                       WorkspaceContext context) {
        if (request == null) throw PortfolioException.validation("analysis request is required");
        projectService.requireProject(projectId, username, context);

        List<Long> requirementIds = selectRequirementIds(projectId, request, username, context);
        int maxNodes = normalizeMaxNodes(request.maxArchitectureNodes());
        if (request.requirementVersionId() != null || request.model() != null) {
            if (request.requirementVersionId() != null && request.analyzeAll()) throw PortfolioException.validation("Select one requirement to analyze a specific version");
            return persistenceService.createOrReuseJob(projectId, requirementIds, normalizeProvider(request.provider()),
                    maxNodes, request.idempotencyKey(), username, context, request.analysisLanguage(),
                    request.requirementVersionId(), request.model());
        }
        if (request.analysisLanguage() != null) return persistenceService.createOrReuseJob(
                projectId, requirementIds, normalizeProvider(request.provider()), maxNodes, request.idempotencyKey(),
                username, context, request.analysisLanguage());
        return persistenceService.createOrReuseJob(
                projectId,
                requirementIds,
                normalizeProvider(request.provider()),
                maxNodes,
                request.idempotencyKey(),
                username,
                context);
    }

    private static AnalyzeProjectRequest requestForRequirement(Long requirementId,
                                                               String provider,
                                                               Integer maxArchitectureNodes,
                                                               String idempotencyKey) {
        return new AnalyzeProjectRequest(
                List.of(requirementId), false, provider, maxArchitectureNodes, idempotencyKey);
    }

    private void dispatch(String jobId,
                          Long projectId,
                          String username,
                          WorkspaceContext context,
                          AnalysisOwner owner) {
        try {
            analysisExecutor.execute(() -> {
                try {
                    executePendingItems(jobId, projectId, username, context, owner);
                } catch (RuntimeException failure) {
                    LOGGER.error("Asynchronous portfolio analysis job {} for project {} stopped unexpectedly",
                            jobId, projectId, failure);
                }
            });
        } catch (TaskRejectedException rejected) {
            throw PortfolioException.unavailable(
                    "Portfolio analysis capacity is currently exhausted; persisted job "
                            + jobId + " can be submitted again",
                    rejected);
        }
    }

    private void requireJobProviderBinding(AnalysisJobView job, String username, WorkspaceContext context) {
        var targets = job.items().stream().map(item -> new ProjectProviderBinding.Target(
                item.requirementId(), item.requirementVersionId())).toList();
        persistenceService.requireRetainedProviderBinding(job.projectId(), job.id(), job.provider(),
                targets, username, context);
    }

    AnalysisOwner admitOwner(String username, WorkspaceContext context) {
        return budgetContinuations == null ? null : budgetContinuations.admitOwner(username, context);
    }

    private AnalysisJobView executePendingItems(String jobId,
                                                Long projectId,
                                                String username,
                                                WorkspaceContext context,
                                                AnalysisOwner owner) {
        String scopeKey = PortfolioScope.key(username, context);
        AnalysisJobView job = persistenceService.getJob(jobId, projectId, username, context);
        var admittedModel = persistenceService.admittedModel(jobId, projectId, username, context);
        String taxonomyFingerprint = admittedModel == null ? fingerprintService.taxonomyFingerprint() : null;
        String promptFingerprint = fingerprintService.promptFingerprint();
        if (job.analysisLanguage() != null) promptFingerprint = com.taxonomy.analysis.recovery.AnalysisCheckpointSession.digest(
                promptFingerprint, com.taxonomy.analysis.service.AnalysisLanguageContext.POLICY, job.analysisLanguage().value());
        String effectiveProvider = job.provider() != null
                ? job.provider() : llmService.getActiveProviderName();

        List<PortfolioAnalysisWorkQueue.WorkItem> workItems = claimPendingItems(jobId, projectId, scopeKey);
        if (workItems.isEmpty()) {
            return reconcileJobState(jobId, projectId, username, context, scopeKey);
        }
        for (PortfolioAnalysisWorkQueue.WorkItem workItem : workItems) {
            String snapshotId = UUID.randomUUID().toString();
            String analysisSessionId = "portfolio:" + snapshotId;
            long startedAt = System.nanoTime();
            try (var providerScope = llmService.withProviderBinding(job.provider(), job.providerBinding());
                 var languageScope = com.taxonomy.analysis.service.AnalysisLanguageContext.open(job.analysisLanguage())) {
                // A rejection after dispatch belongs to this claim's normal failure and reconciliation path.
                persistenceService.requireRetainedProviderBinding(workItem.projectId(), workItem.jobId(), job.provider(),
                        List.of(new ProjectProviderBinding.Target(workItem.requirementId(), workItem.requirementVersionId())),
                        username, context);
                AnalyzeRequirementCommand command = new AnalyzeRequirementCommand(
                                workItem.requirementText(),
                                true,
                                job.maxArchitectureNodes(),
                                job.provider(),
                                username,
                                context,
                                new AnalysisProvenance(
                                        projectId,
                                        workItem.requirementId(),
                                        snapshotId,
                                        analysisSessionId, workItem.requirementVersionId()),
                                admittedModel == null ? com.taxonomy.dto.AnalysisScope.full() : admittedModel.scope(),
                                job.providerBinding())
                                .withAnalysisReasonLength(job.analysisReasonLength() == null
                                        ? com.taxonomy.dto.AnalysisReasonLength.STANDARD : job.analysisReasonLength())
                                .withAnalysisLanguage(job.analysisLanguage())
                                .withModelBinding(admittedModel);
                AnalysisResult analysis;
                // Close the question journal before optional snapshot enrichment. Only
                // the actual classification/relation procedure owns these checkpoints.
                try (var continuation = budgetContinuations == null ? null
                        : budgetContinuations.begin(workItem, job, username, context, admittedModel, owner)) {
                    if (continuation == null) {
                        analysis = analyzeRequirementUseCase.analyze(command).analysisResult();
                    } else if (continuation.replay() != null) {
                        analysis = continuation.replay();
                    } else {
                        var result = analyzeRequirementUseCase.analyze(continuation.bind(command)).analysisResult();
                        if (owner != null) budgetContinuations.checkOwner(owner, username, context);
                        analysis = continuation.complete(result);
                    }
                }
                if (analysis == null || analysis.getScores() == null
                        || "ERROR".equalsIgnoreCase(analysis.getStatus())) {
                    throw new IllegalStateException(analysis != null && analysis.getErrorMessage() != null
                            ? analysis.getErrorMessage() : "Analysis produced no usable scores");
                }

                AnalysisCopilotResult copilot = analysis.getCopilot();
                if (copilot == null) {
                    // Historical/local producers may not have a report yet. Use their
                    // retained evidence, never a newer global catalogue or model call.
                    copilot = copilotService.analyze(analysis, workItem.requirementText());
                    analysis.setCopilot(copilot);
                }
                GapAnalysisView gaps = copilot.gaps();
                PatternDetectionView patterns = copilot.patterns();
                ArchitectureRecommendation recommendation = copilot.recommendation();
                long durationMs = (System.nanoTime() - startedAt) / 1_000_000L;
                if (owner != null) budgetContinuations.checkOwner(owner, username, context);
                claimPersistenceService.persistSnapshot(
                        workItem,
                        snapshotId,
                        analysisSessionId,
                        analysis,
                        gaps,
                        patterns,
                        recommendation,
                        effectiveProvider,
                        null,
                        promptFingerprint,
                        admittedModel == null ? taxonomyFingerprint
                                : com.taxonomy.dto.TaxonomyDataFingerprint.sha256(analysis.getTree()),
                        username,
                        context,
                        durationMs);
            } catch (Exception failure) {
                try {
                    claimPersistenceService.failItem(workItem, failure);
                } catch (PortfolioException claimFailure) {
                    if (claimFailure.getKind() != PortfolioException.Kind.CONFLICT) {
                        throw claimFailure;
                    }
                    LOGGER.info(
                            "Discarded late result for analysis item {} attempt {} because its claim is no longer active",
                            workItem.itemId(),
                            workItem.attempt());
                }
            }
        }
        return reconcileJobState(jobId, projectId, username, context, scopeKey);
    }

    private List<PortfolioAnalysisWorkQueue.WorkItem> claimPendingItems(String jobId, Long projectId, String scopeKey) {
        // Each call starts a new transaction. Only retry a rolled-back claim, before any provider work.
        for (int attempt = 1; ; attempt++) {
            try {
                return workQueue.pending(jobId, projectId, scopeKey);
            } catch (org.springframework.dao.PessimisticLockingFailureException contention) {
                if (attempt == MAX_CLAIM_TRANSACTION_ATTEMPTS) throw contention;
            }
        }
    }

    private AnalysisJobView reconcileJobState(String jobId,
                                              Long projectId,
                                              String username,
                                              WorkspaceContext context,
                                              String scopeKey) {
        AnalysisJobView current = persistenceService.getJob(
                jobId, projectId, username, context);
        if (isTerminal(current.status())) {
            return current;
        }
        if (recoveryService.markPendingWhenOnlyPreparedItemsRemain(
                jobId, projectId, scopeKey)) {
            return persistenceService.getJob(jobId, projectId, username, context);
        }
        return persistenceService.completeJob(jobId, projectId, scopeKey);
    }

    private static boolean isTerminal(AnalysisStatus status) {
        return status == AnalysisStatus.SUCCESS
                || status == AnalysisStatus.PARTIAL
                || status == AnalysisStatus.FAILED
                || status == AnalysisStatus.CANCELLED;
    }

    private List<Long> selectRequirementIds(Long projectId,
                                            AnalyzeProjectRequest request,
                                            String username,
                                            WorkspaceContext context) {
        if (request.analyzeAll()) {
            List<Long> ids = projectService.listRequirements(projectId, username, context).stream()
                    .map(requirement -> requirement.id())
                    .toList();
            if (ids.isEmpty()) throw PortfolioException.validation("Project has no requirements to analyze");
            return requireAllowedBatch(ids);
        }
        if (request.requirementIds() == null || request.requirementIds().isEmpty()) {
            throw PortfolioException.validation("requirementIds must be supplied unless all=true");
        }
        List<Long> ids = request.requirementIds().stream()
                .filter(java.util.Objects::nonNull)
                .distinct()
                .toList();
        if (ids.isEmpty()) {
            throw PortfolioException.validation("At least one valid requirementId is required");
        }
        return requireAllowedBatch(ids);
    }

    private List<Long> requireAllowedBatch(List<Long> requirementIds) {
        if (requirementIds.size() > maximumBatchRequirements) {
            throw PortfolioException.validation(
                    "Analysis batch contains " + requirementIds.size()
                            + " requirements; maximum is " + maximumBatchRequirements);
        }
        return requirementIds;
    }

    private int normalizeMaxNodes(Integer requested) {
        int value = requested != null ? requested : DEFAULT_ARCHITECTURE_NODES;
        if (value < 1 || value > maximumArchitectureNodes) {
            throw PortfolioException.validation(
                    "maxArchitectureNodes must be between 1 and " + maximumArchitectureNodes);
        }
        return value;
    }

    private static String normalizeProvider(String provider) {
        if (provider == null || provider.isBlank()) return null;
        return provider.strip().toUpperCase(Locale.ROOT);
    }

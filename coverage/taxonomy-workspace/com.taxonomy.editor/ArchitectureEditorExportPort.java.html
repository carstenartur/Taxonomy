package com.taxonomy.editor;

import com.taxonomy.diagram.DiagramCanonicalEvidence;
import com.taxonomy.diagram.DiagramScene;
import com.taxonomy.workspace.service.RepositoryContext;

import java.io.IOException;
import java.util.Objects;

/**
 * Workspace/editor-owned read boundary for exact, render-ready architecture exports.
 *
 * <p>The port exposes canonical source or neutral scene data with provenance. HTTP composition and
 * concrete SVG/PDF rendering remain outside the workspace authority boundary.</p>
 */
public interface ArchitectureEditorExportPort {

    ExportDocument read(RepositoryContext context, String commit, Long revision) throws IOException;

    CheckpointModel readCheckpoint(RepositoryContext context, String commit) throws IOException;

    record CheckpointModel(String commit, String dsl) { }

    default ExportDocument read(RepositoryContext context, String commit, Long revision,
            com.taxonomy.diagram.DiagramLayout.Strategy strategy, Long layoutRevision) throws IOException {
        if (strategy != com.taxonomy.diagram.DiagramLayout.Strategy.STABLE || layoutRevision != null)
            throw new IllegalArgumentException("This export adapter does not support the selected layout");
        return read(context, commit, revision);
    }

    default ExportDocument read(RepositoryContext context, String commit, Long revision,
            com.taxonomy.diagram.DiagramLayout.Strategy strategy, Long layoutRevision, String viewId) throws IOException {
        if (viewId != null && !viewId.isBlank()) throw new IllegalArgumentException("This export adapter does not support selected views");
        return read(context, commit, revision, strategy, layoutRevision);
    }

    record ExportDocument(
            String repositoryId,
            String workspaceScopeKey,
            String branch,
            String commit,
            long revision,
            String source,
            String layoutMode,
            DiagramScene scene,
            DiagramCanonicalEvidence evidence) {
        public ExportDocument(String repositoryId, String workspaceScopeKey, String branch, String commit,
                long revision, String source, String layoutMode, DiagramScene scene) {
            this(repositoryId, workspaceScopeKey, branch, commit, revision, source, layoutMode, scene,
                    new DiagramCanonicalEvidence(null, java.util.List.of(), java.util.Map.of()));
        }

        public ExportDocument {
            repositoryId = Objects.requireNonNull(repositoryId, "repositoryId");
            workspaceScopeKey = Objects.requireNonNull(workspaceScopeKey, "workspaceScopeKey");
            branch = Objects.requireNonNull(branch, "branch");
            source = Objects.requireNonNull(source, "source");
            layoutMode = Objects.requireNonNull(layoutMode, "layoutMode");
            scene = Objects.requireNonNull(scene, "scene");
            evidence = Objects.requireNonNull(evidence, "evidence");
        }
    }
}
